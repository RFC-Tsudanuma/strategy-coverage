71def tackle_ball( 

72 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

73) -> StrategyState | None: 

74 """ 

75 ボールの位置に突っ込む。相手がいるいないは特に関係なし。 

76 ボールに向かっていく進行方向そのままに蹴る。 

77 蹴る方向等は歩き始めの位置くらいでしか調整できないため、クリア等のために使う 

78 """ 

79 # preconditionを確認 

80 if s.world_states["ball_position_relative"] != BALL_POSITION.UNKNOWN: 

81 print_in_real_action(" Executing tackle_ball().") 

82 obstacles = get_all_obstacles(s) 

83 planned_path, _ = pathplan.find_path_to_target( 

84 s.ball_position_local, 

85 obstacles, 

86 clearance=setting.DEFAULT_STRATEGY_CONFIG["common_config"][ 

87 "pathplan_clearance" 

88 ], 

89 clearance_factor=pathplan.distance_clearance_factor( 

90 s.ball_position_local, 

91 obstacles, 

92 clearance=setting.DEFAULT_STRATEGY_CONFIG["common_config"][ 

93 "pathplan_clearance" 

94 ], 

95 ), 

96 publisher=pub, 

97 ) 

98 if not action_functions.validate_path(planned_path): 

99 # パスが見つからない場合、停止 

100 pub.send_walk_command(WalkCommand(0.0, 0.0, 0.0)) 

101 logger.action_executed(details="No valid path found. Sent stop command.") 

102 return s 

103 cmd = action_functions.track_planned_path(planned_path, pub) 

104 pub.send_walk_command(cmd) 

105 # デバック用 

106 pub.send_target_position( 

107 convert_local_to_global(s.self_position_global, s.ball_position_local) 

108 ) 

109 logger.action_executed(details="Tackling the ball") 

110 return s 

111 else: 

112 print_in_real_action("Ball is not in a position to approach.") 

113 logger.action_precondition_not_satisfied(details="Ball position unknown") 

114 return None

254@contract.htn_action() 

255def turn_to_face_ball_and_goal( 

256 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

257) -> StrategyState | None: 

258 """ 

259 体の正面をボールとゴール中心への単位ベクトルの合成方向に向ける。 

260 """ 

261 print_in_real_action("Executing turn_to_face_ball_and_goal().") 

262 vec_ball = np.array([s.ball_position_local.x, s.ball_position_local.y], dtype=float) 

263 goal_local = get_goal_center_pos(s) 

264 vec_goal = np.array([goal_local.x, goal_local.y], dtype=float) 

265 

266 norm_ball = np.linalg.norm(vec_ball) 

267 norm_goal = np.linalg.norm(vec_goal) 

268 if norm_ball < 1e-6 or norm_goal < 1e-6: 

269 logger.action_executed( 

270 details="Cannot compute direction (zero vector). Do nothing." 

271 ) 

272 return s 

273 

274 # ボールとゴール中心への単位ベクトルを足してシュート時に体を向けるべき方向を決める 

275 vec_combined = vec_ball / norm_ball + vec_goal / norm_goal 

276 combined_target = Position( 

277 Position.COORD_LOCAL, float(vec_combined[0]), float(vec_combined[1]) 

278 ) 

279 

280 if is_body_facing_target(combined_target): 

281 logger.action_postcondition_already_satisfied( 

282 details="Already facing ball-goal combined direction" 

283 ) 

284 print_in_real_action("Already facing ball-goal combined direction.") 

285 return s 

286 

287 cmd = action_functions.generate_turn_command(combined_target) 

288 pub.send_walk_command(cmd) 

289 logger.action_executed(details="Turning body to ball-goal combined direction") 

290 return s
