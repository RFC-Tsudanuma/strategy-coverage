25"""GUI ツール向けに strategy_shared_config.yaml のフィールド寸法(cm)をメートルで提供する。 

26 

27戦略本体は setting.FIELD_DIMENTIONS (cm) を使うが、描画系のツールはメートル基準で 

28座標変換しているため、ここで単位変換したものを共通の入口として提供する。 

29""" 

30 

31from setting import STRATEGY_SHARED_CONFIG 

32 

33# config にキーが無い場合のフォールバック(メートル)。middleサイズのフィールド相当。 

34_DEFAULT_FIELD_DIMENSIONS_M = { 

35 "field_length": 14.0, 

36 "field_width": 9.0, 

37 "penalty_area_length": 3.0, 

38 "penalty_area_width": 6.0, 

39 "goal_area_length": 1.0, 

40 "goal_area_width": 4.0, 

41 "penalty_spot_distance": 2.1, 

42 "goalpost_width": 2.6, 

43 "goal_depth": 0.6, 

44 "center_circle_radius": 1.5, 

45} 

46 

47 

48def load_field_dimensions_m() -> dict[str, float]: 

49 """strategy_shared_config.yaml の field_config (cm) を読み、メートル単位の 

50 フィールド寸法 dict を返す。config に無いキーは既定値にフォールバックする。 

51 """ 

52 field_config = STRATEGY_SHARED_CONFIG.get("field_config", {}) or {} 

53 

54 def _m(key: str) -> float: 

55 value_cm = field_config.get(key) 

56 if value_cm is None: 

57 return _DEFAULT_FIELD_DIMENSIONS_M[key] 

58 return float(value_cm) / 100.0 

59 

60 # センターサークルは config に直径(cm)で持つため半径(m)へ変換する。 

61 center_circle_diameter_cm = field_config.get("center_circle_diameter") 

62 center_circle_radius = ( 

63 float(center_circle_diameter_cm) / 100.0 / 2.0 

64 if center_circle_diameter_cm is not None 

65 else _DEFAULT_FIELD_DIMENSIONS_M["center_circle_radius"] 

66 ) 

67 return { 

68 "field_length": _m("field_length"), 

69 "field_width": _m("field_width"), 

70 "penalty_area_length": _m("penalty_area_length"), 

71 "penalty_area_width": _m("penalty_area_width"), 

72 "goal_area_length": _m("goal_area_length"), 

73 "goal_area_width": _m("goal_area_width"), 

74 "penalty_spot_distance": _m("penalty_spot_distance"), 

75 "goalpost_width": _m("goalpost_width"), 

76 "goal_depth": _m("goal_depth"), 

77 "center_circle_radius": center_circle_radius, 

78 }
