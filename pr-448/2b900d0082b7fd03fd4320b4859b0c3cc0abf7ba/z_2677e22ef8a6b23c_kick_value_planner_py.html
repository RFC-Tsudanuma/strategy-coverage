1"""policy kick のキック方向を、計測したキック方向の誤差の分布 (kick_value_cpp) で選ぶ。 

2 

3モデル (旋回量ごとのキック方向の誤差など) は config/kick_value_model.yaml にあり、 

4tools/kick_value_model_from_trials.py が sim の計測 (遅延のある観測) から作る。 

5kick_value_config.mode で使い方を選ぶ: 

6 shot: 毎回ゴールを狙い、旋回量ごとの誤差の中央値を見込んで、ボールがゴールの真ん中へ 

7 向かうように指令をずらす (KickShot) 

8 plan: 価値反復で、途中のキックも含めて得点までの時間が最短の向きを選ぶ。価値表は 

9 戦略の起動後に別スレッドで解き、解けるまで plan() は None を返すので、 

10 呼び出し側はゴール中央へ蹴る従来の動きに戻る (KickValuePlanner) 

11 

12蹴り足: kick_value_config.foot_selection が both なら、状況ごとに左右の足のうち良い方を選び 

13(KickPlan.foot)、fixed (既定) なら設定 kick_foot の足だけで考える。モデルの誤差は左足のもので、 

14右足はその鏡写し (kick_value_iteration.hpp)。 

15 

16座標は「自陣が左 (相手ゴールが +x)」の視点 [m] で解く。自陣が右のときはフィールドの中心まわりの 

17180° 回転 (x, y, θ) -> (-x, -y, θ + π) で行き来する。鏡映 (x -> -x) だと左右が入れ替わり、 

18反時計回りの偏り dir_bias や蹴り足の左右が逆になる。 

19"""

38@dataclass(frozen=True) 

39class KickPlan:

47def _kick_foot(foot: int) -> KICK_FOOT: 

48 """kick_value_cpp の LEFT_FOOT / RIGHT_FOOT を KICK_FOOT にする。""" 

49 return KICK_FOOT.RIGHT if foot == kick_value_cpp.RIGHT_FOOT else KICK_FOOT.LEFT

52FOOT_NAMES = {"left": kick_value_cpp.LEFT_ONLY, "right": kick_value_cpp.RIGHT_ONLY}

55def foot_selection(kick_value_cfg: dict[str, Any], kick_foot: str) -> int: 

56 """kick_value_config.foot_selection (fixed / both) と設定 kick_foot (left / right) から 

57 kick_value_cpp の foot_selection。""" 

58 sel = kick_value_cfg.get("foot_selection", "fixed") 

59 if sel == "both": 

60 return kick_value_cpp.BOTH_FEET 

61 if sel != "fixed": 

62 raise ValueError( 

63 f"kick_value_config.foot_selection must be fixed or both: {sel!r}" 

64 ) 

65 if kick_foot not in FOOT_NAMES: 

66 raise ValueError(f"kick_foot must be right or left: {kick_foot!r}") 

67 return FOOT_NAMES[kick_foot]

70def build_model( 

71 model_cfg: dict[str, Any], 

72 field: dict[str, float], 

73 feet: int = kick_value_cpp.LEFT_ONLY, 

74) -> kick_value_cpp.Model: 

75 """kick_value_model.yaml の中身とフィールド寸法 [m] から C++ のモデルを作る。 

76 feet: kick_value_cpp の foot_selection (LEFT_ONLY / RIGHT_ONLY / BOTH_FEET)。""" 

77 m = kick_value_cpp.Model() 

78 m.field_length = field["field_length"] 

79 m.field_width = field["field_width"] 

80 m.goal_half_width = field["goalpost_width"] / 2.0 - model_cfg["ball_radius_m"] 

81 m.grid_res = model_cfg["grid_res_m"] 

82 m.n_heading = model_cfg["n_heading"] 

83 m.n_dir = model_cfg["n_dir"] 

84 m.turn_edges = [math.radians(e) for e in model_cfg["turn_edges_deg"]] 

85 m.dir_err = [[math.radians(e) for e in b] for b in model_cfg["dir_err_deg"]] 

86 m.dir_err_cw = [ 

87 [math.radians(e) for e in b] for b in model_cfg.get("dir_err_cw_deg", []) 

88 ] 

89 m.dir_bias = math.radians(model_cfg.get("dir_bias_deg", 0.0)) 

90 m.kick_time = list(model_cfg["kick_time_s"]) 

91 sigma = model_cfg["lateral_sigma_m"] 

92 if sigma > 0.0: 

93 # 3 点 Gauss-Hermite (正規分布の 5 次までのモーメントが一致する) 

94 m.lateral = [-math.sqrt(3.0) * sigma, 0.0, math.sqrt(3.0) * sigma] 

95 m.lateral_w = [1.0 / 6.0, 2.0 / 3.0, 1.0 / 6.0] 

96 else: 

97 m.lateral = [0.0] 

98 m.lateral_w = [1.0] 

99 m.roll = list(model_cfg["roll_m"]) 

100 m.walk_speed = model_cfg["walk_speed_mps"] 

101 m.out_penalty = model_cfg["out_penalty_s"] 

102 m.own_goal_penalty = model_cfg["own_goal_penalty_s"] 

103 m.align_time_per_rad = model_cfg["align_time_per_rad_s"] 

104 m.foot_selection = feet 

105 m.validate() 

106 return m

109class Planner(Protocol): 

110 name: str 

111 

112 def plan( 

113 self, ball_xy: tuple[float, float], heading: float, mirrored: bool 

114 ) -> KickPlan | None: ... 

115 

116 

117def _to_solver_frame( 

118 ball_xy: tuple[float, float], heading: float, mirrored: bool 

119) -> tuple[float, float, float]: 

120 x, y = ball_xy 

121 return (-x, -y, heading + math.pi) if mirrored else (x, y, heading) 

122 

123 

124def _from_solver_frame(direction: float, mirrored: bool) -> float: 

125 d = direction + math.pi if mirrored else direction 

126 return math.atan2(math.sin(d), math.cos(d))

129class KickShot: 

130 """毎回ゴールを狙い、誤差の中央値を見込んでゴールの真ん中へ蹴る (解く表は無い)。""" 

131 

132 name = "goal_shot" 

133 

134 def __init__(self, model: kick_value_cpp.Model) -> None: 

135 self._shot = kick_value_cpp.Shot(model) 

136 

137 def plan( 

138 self, ball_xy: tuple[float, float], heading: float, mirrored: bool 

139 ) -> KickPlan: 

140 x, y, heading = _to_solver_frame(ball_xy, heading, mirrored) 

141 d, p_goal, foot = self._shot.aim_best(x=x, y=y, heading=heading) 

142 return KickPlan( 

143 direction=_from_solver_frame(d, mirrored), 

144 aligned=False, 

145 expected_time=math.nan, 

146 p_goal=p_goal, 

147 foot=_kick_foot(foot), 

148 )

151class KickValuePlanner: 

152 name = "value_iteration" 

153 

154 def __init__( 

155 self, model: kick_value_cpp.Model, threads: int, max_iter: int, tol: float 

156 ) -> None: 

157 self._table = kick_value_cpp.Table(model) 

158 self._ready = threading.Event() 

159 self._thread = threading.Thread( 

160 target=self._solve, args=(max_iter, tol, threads), daemon=True 

161 ) 

162 self._thread.start() 

163 

164 def _solve(self, max_iter: int, tol: float, threads: int) -> None: 

165 self._table.solve(max_iter=max_iter, tol=tol, threads=threads) 

166 self._ready.set() 

167 

168 @property 

169 def ready(self) -> bool: 

170 return self._ready.is_set() 

171 

172 def wait(self, timeout: float | None = None) -> bool: 

173 return self._ready.wait(timeout) 

174 

175 def plan( 

176 self, ball_xy: tuple[float, float], heading: float, mirrored: bool 

177 ) -> KickPlan | None: 

178 """ボール位置 [m] とロボットの向き [rad] (戦略のワールド座標) から最善のキック。 

179 mirrored: 自陣が右 (相手ゴールが -x)。解けていなければ None。""" 

180 if not self.ready: 

181 return None 

182 x, y, heading = _to_solver_frame(ball_xy, heading, mirrored) 

183 d, q, p_goal, aligned, foot = self._table.best( 

184 x=x, y=y, heading=heading, dirs=DIRECTIONS 

185 ) 

186 return KickPlan( 

187 direction=_from_solver_frame(d, mirrored), 

188 aligned=bool(aligned), 

189 expected_time=-q, 

190 p_goal=p_goal, 

191 foot=_kick_foot(foot), 

192 )

195class _KickPlanner: 

196 """get_planner() が作ったプランナの置き場。""" 

197 

198 planner: Planner | None = None 

199 lock = threading.Lock() 

200 

201 

202_holder = _KickPlanner() 

203 

204 

205def get_planner( 

206 kick_value_cfg: dict[str, Any], 

207 config_dir: str, 

208 field: dict[str, float], 

209 kick_foot: str = "left", 

210) -> Planner | None: 

211 """共有設定の kick_value_config が有効なら、プロセスで 1 つのプランナを返す。 

212 kick_foot: 共有設定の kick_foot (foot_selection: fixed のときに蹴る足)。""" 

213 if not kick_value_cfg.get("enable", False): 

214 return None 

215 with _holder.lock: 

216 if _holder.planner is None: 

217 path = os.path.join(config_dir, kick_value_cfg["model_file"]) 

218 with open(path, encoding="utf-8") as f: 

219 model = build_model( 

220 yaml.safe_load(f), field, foot_selection(kick_value_cfg, kick_foot) 

221 ) 

222 mode = kick_value_cfg["mode"] 

223 if mode == "shot": 

224 _holder.planner = KickShot(model) 

225 elif mode == "plan": 

226 _holder.planner = KickValuePlanner( 

227 model, 

228 threads=kick_value_cfg["solve_threads"], 

229 max_iter=kick_value_cfg["solve_max_iter"], 

230 tol=kick_value_cfg["solve_tol_s"], 

231 ) 

232 else: 

233 raise ValueError( 

234 f"kick_value_config.mode must be shot or plan: {mode!r}" 

235 ) 

236 return _holder.planner
