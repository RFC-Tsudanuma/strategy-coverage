39def get_most_reliable_shared_ballpos( 

40 shared_info: list[state.SharedMessage], 

41) -> state.Position | None: 

42 """ 

43 共有された情報の中から、最も信頼できるボールの情報を取得する 

44 """ 

45 if shared_info is None: 

46 return None 

47 most_reliable_ballpos = None 

48 max_distance = 1e-6 

49 for info in shared_info: 

50 if info is None: 

51 continue 

52 if info.ballpos is not None and info.selfpos is not None: 

53 if most_reliable_ballpos is None: 

54 most_reliable_ballpos = info.ballpos 

55 else: 

56 current_distance = calculate_distance(info.ballpos_local) 

57 if current_distance > max_distance: 

58 most_reliable_ballpos = info.ballpos 

59 max_distance = current_distance 

60 return most_reliable_ballpos

63def get_most_reliable_ballpos_global( 

64 current_selfpos: state.Position | None, 

65 current_detected_ballpos_local: state.Position | None, 

66 shared_info: list[state.SharedMessage], 

67) -> state.Position | None: 

68 """ 

69 共有された情報と今持っている情報の中から、最も信頼できるボールの情報を取得する 

70 ボールとの距離が遠い = ピッチ角が高い = 自己位置推定が正しい確率が高い、なので、ボールとの距離が一番遠いやつを信頼する。 

71 """ 

72 if shared_info is None: 

73 return None 

74 most_reliable_ballpos = None 

75 max_distance = 1e-6 

76 # ひとまず自分の観測にする 

77 if current_detected_ballpos_local is not None and current_selfpos is not None: 

78 most_reliable_ballpos = convert_local_to_global( 

79 current_selfpos, current_detected_ballpos_local 

80 ) 

81 max_distance = calculate_distance(current_detected_ballpos_local) 

82 # 受け取った情報と比較する 

83 for info in shared_info: 

84 if info is None: 

85 continue 

86 if info.ballpos is not None and info.selfpos is not None: 

87 if most_reliable_ballpos is None: 

88 most_reliable_ballpos = info.ballpos 

89 else: 

90 current_distance = calculate_distance(info.ballpos_local) 

91 if current_distance > max_distance: 

92 most_reliable_ballpos = info.ballpos 

93 max_distance = current_distance 

94 return most_reliable_ballpos

125def get_other_player_average_ballpos_from_shared_info( 

126 shared_info: list[state.SharedMessage | None], 

127) -> state.Position | None: 

128 """ 

129 SharedMessageのリストから有効なボール位置を取り出す 

130 """ 

131 ballpos = state.Position(x=0.0, y=0.0, coord=state.Position.COORD_GLOBAL) 

132 count = 0 

133 for info in shared_info: 

134 if info is not None and info.ballpos is not None: 

135 ballpos.x += info.ballpos.x 

136 ballpos.y += info.ballpos.y 

137 count += 1 

138 if count > 0: 

139 ballpos.x /= count 

140 ballpos.y /= count 

141 return ballpos 

142 else: 

143 return None

203def shared_info_is_within_second(shared_info: state.SharedMessage, seconds: float) -> bool: 

204 """ 

205 共有された情報が現在時刻から指定した秒数以内のものであるかを判定する 

206 """ 

207 if shared_info is None or shared_info.send_timestamp_ms == 0: 

208 return False 

209 current_time_ms = datetime.datetime.now(datetime.timezone.utc).timestamp() * 1000 

210 time_diff = (current_time_ms - shared_info.send_timestamp_ms) 

211 if time_diff < 0: 

212 return False 

213 return time_diff <= seconds * 1000
