74def remember_enemy_seen_by_self( 

75 s: StrategyState, enemy_local: Position, now: float 

76) -> None: 

77 """ 

78 自分のカメラで見た敵を s.enemies_seen_by_self に覚える。 

79 merge 距離以内の記憶は同じ敵とみなして位置と時刻を更新し、古い記憶は捨てる。 

80 """ 

81 cfg = STRATEGY_SHARED_CONFIG["course_cut_cover"] 

82 enemy_global = convert_local_to_global(s.self_position_global, enemy_local) 

83 s.enemies_seen_by_self = [ 

84 (pos, seen_at) 

85 for pos, seen_at in s.enemies_seen_by_self 

86 if now - seen_at <= cfg["enemies_seen_stale_sec"] 

87 and math.hypot(pos.x - enemy_global.x, pos.y - enemy_global.y) 

88 > cfg["enemies_seen_merge_distance_cm"] 

89 ] 

90 s.enemies_seen_by_self.append((enemy_global, now))

93def enemies_seen_by_self_except_kicker(s: StrategyState, now: float) -> list[Position]: 

94 """ 

95 自分のカメラで最近 (enemies_seen_stale_sec 以内) 見た敵 (グローバル座標)。 

96 ボールから SET_PLAY_KEEP_AWAY_DISTANCE 以内の敵 (セットプレーのキッカー) は除く。 

97 キッカーはボールを見ている味方が共有マップに入れるので、自分で探す必要が無い。 

98 共有マップ (JPDA) と同じく、フィールドの外 (ベンチや退場中のロボット) の敵も除く。 

99 """ 

100 stale_sec = STRATEGY_SHARED_CONFIG["course_cut_cover"]["enemies_seen_stale_sec"] 

101 ball = s.common_map_obs.ball_position 

102 if ball is None and s.ball_position_local is not None: 

103 ball = convert_local_to_global(s.self_position_global, s.ball_position_local) 

104 return [ 

105 pos 

106 for pos, seen_at in s.enemies_seen_by_self 

107 if now - seen_at <= stale_sec 

108 and not is_position_outside_field_bounds( 

109 pos, FIELD_DIMENTIONS.OPPONENT_IGNORE_OUT_OF_FIELD_MARGIN 

110 ) 

111 and ( 

112 ball is None 

113 or math.hypot(pos.x - ball.x, pos.y - ball.y) > SET_PLAY_KEEP_AWAY_DISTANCE 

114 ) 

115 ]

153def goal_net_walls( 

154 goalposts_local: list[Position], 

155 self_position_global: Position, 

156 goalpost_width: float, 

157 goal_depth: float, 

158) -> list[Wall]: 

159 """ 

160 1 つのゴールのポストから、そのゴールのネットを表すコの字の壁 (横 2 本と奥 1 本) を作る 

161 

162 最寄りのポストを near とし、幅方向に goalpost_width 進んだ点を far とする。 

163 幅方向は次に近いポストへの向き。見えてないなら global_y に沿うゴール中心向き。 

164 """ 

165 if not goalposts_local: 

166 return [] 

167 by_distance = sorted(goalposts_local, key=calculate_distance) 

168 near_net_point = convert_local_to_global(self_position_global, by_distance[0]) 

169 

170 # far_net_point 

171 goal_line_unit = (0.0, -math.copysign(1.0, near_net_point.y)) 

172 if len(by_distance) >= 2: 

173 # 2 本見えたら、near_net_point から 2 番目に近いポストへ向く単位ベクトル 

174 far_post = convert_local_to_global(self_position_global, by_distance[1]) 

175 direction = get_unit_vec_between(far_post, near_net_point) 

176 if (direction.x, direction.y) != (0.0, 0.0): 

177 goal_line_unit = (direction.x, direction.y) 

178 far_net_point = ( 

179 near_net_point.x + goal_line_unit[0] * goalpost_width, 

180 near_net_point.y + goal_line_unit[1] * goalpost_width, 

181 ) 

182 

183 goal_normal_unit = (-goal_line_unit[1], goal_line_unit[0]) 

184 if goal_normal_unit[0] * math.copysign(1.0, near_net_point.x) < 0: 

185 goal_normal_unit = (-goal_normal_unit[0], -goal_normal_unit[1]) 

186 

187 net_points = [ 

188 (near_net_point.x, near_net_point.y), 

189 ( 

190 near_net_point.x + goal_normal_unit[0] * goal_depth, 

191 near_net_point.y + goal_normal_unit[1] * goal_depth, 

192 ), 

193 ( 

194 far_net_point[0] + goal_normal_unit[0] * goal_depth, 

195 far_net_point[1] + goal_normal_unit[1] * goal_depth, 

196 ), 

197 far_net_point, 

198 ] 

199 local_net_points = [ 

200 convert_global_to_local( 

201 self_position_global, Position(Position.COORD_GLOBAL, x, y) 

202 ) 

203 for x, y in net_points 

204 ] 

205 return [Wall(start, end) for start, end in itertools.pairwise(local_net_points)]

208def get_pathplan_goalposts( 

209 current_state: StrategyState, 

210) -> tuple[list[Position], list[Position]]: 

211 """ 

212 経路計画に使うゴールポスト (ローカル座標) を (敵ゴール, 味方ゴール) で返す。 

213 1 本も見えていないゴールは、自己位置とフィールドの寸法から求めた 2 本のうち近い方が 

214 pathplan_field_goalpost_range_cm 以内のときだけ、その 2 本を使う。 

215 """ 

216 max_range = STRATEGY_SHARED_CONFIG["pathplan_field_goalpost_range_cm"] 

217 

218 def field_goalposts(goal_line_x: float) -> list[Position]: 

219 posts = [ 

220 convert_global_to_local(current_state.self_position_global, post) 

221 for post in get_field_goalposts_global(current_state, goal_line_x) 

222 ] 

223 if min(calculate_distance(post) for post in posts) > max_range: 

224 return [] 

225 return posts 

226 

227 goal_line_x = FIELD_DIMENTIONS.X_LENGTH / 2.0 

228 return ( 

229 current_state.opponent_goalpost_local or field_goalposts(goal_line_x), 

230 current_state.ally_goalpost_local or field_goalposts(-goal_line_x), 

231 )

234def get_pathplan_obstacles(current_state: StrategyState) -> list[Obstacle]: 

235 """ 

236 現在の状態から、経路計画に渡す障害物 (位置と keepout 半径) を取得する 

237 """ 

238 opponent_goalposts, ally_goalposts = get_pathplan_goalposts(current_state) 

239 robot = PATHPLAN_CONFIG.robot_keepout_radius_cm 

240 goalpost = PATHPLAN_CONFIG.goalpost_keepout_radius_cm 

241 positions_with_radius = ( 

242 (current_state.opponent_position_local, robot), 

243 (current_state.allies_position_local, robot), 

244 (opponent_goalposts, goalpost), 

245 (ally_goalposts, goalpost), 

246 (current_state.normal_obstacles_position_local, robot), 

247 ) 

248 obstacles: list[Obstacle] = [] 

249 for positions, keepout_radius_cm in positions_with_radius: 

250 for position in positions: 

251 obstacles.append(Obstacle(position, keepout_radius_cm)) 

252 return obstacles

270def circle_avoid_zone(center_local: Position, radius: float) -> AvoidZone: 

271 """ 

272 中心 center_local・半径 radius の円に外接する正多角形の範囲を作る (ローカル座標、cm)。 

273 """ 

274 # 円に外接する正多角形の頂点数 

275 _CIRCLE_AVOID_ZONE_VERTICES = 8 

276 

277 vertex_radius = radius / math.cos(math.pi / _CIRCLE_AVOID_ZONE_VERTICES) 

278 vertices = [] 

279 for k in range(_CIRCLE_AVOID_ZONE_VERTICES): 

280 angle = 2.0 * math.pi * k / _CIRCLE_AVOID_ZONE_VERTICES 

281 x = center_local.x + vertex_radius * math.cos(angle) 

282 y = center_local.y + vertex_radius * math.sin(angle) 

283 vertices.append(Position(Position.COORD_LOCAL, x, y)) 

284 return AvoidZone(tuple(vertices))

287def global_rectangle_avoid_zone( 

288 current_state: StrategyState, 

289 x_min: float, 

290 x_max: float, 

291 y_min: float, 

292 y_max: float, 

293) -> AvoidZone: 

294 """ 

295 グローバル座標から長方形の範囲を作る (ローカル座標)。 

296 """ 

297 corners = ( 

298 (x_min, y_min), 

299 (x_max, y_min), 

300 (x_max, y_max), 

301 (x_min, y_max), 

302 ) 

303 vertices = [] 

304 for x, y in corners: 

305 corner_global = convert_hardcoded_pos_to_our_field( 

306 current_state, Position(Position.COORD_GLOBAL, x, y) 

307 ) 

308 vertices.append( 

309 convert_global_to_local(current_state.self_position_global, corner_global) 

310 ) 

311 return AvoidZone(tuple(vertices))

314def get_pathplan_avoid_zones(current_state: StrategyState) -> list[AvoidZone]: 

315 """ 

316 経路計画に渡す、避ける範囲を取得する (ローカル座標、cm)。 

317 相手チームのセットプレー中に、入れない範囲を返す。 

318 ゴールキックは相手のペナルティエリア全体、それ以外はボールから SET_PLAY_KEEP_AWAY_DISTANCE 以内。 

319 ボールが見えていなければボールの範囲は作らない。 

320 """ 

321 set_play = current_state.world_states["set_play"] 

322 if set_play not in SET_PLAY_STATES or current_state.kickOffTeam == get_team_id(): 

323 return [] 

324 if set_play == SET_PLAY.GOAL_KICK: 

325 half_width = FIELD_DIMENTIONS.PENALTY_AREA_WIDTH / 2.0 

326 return [ 

327 global_rectangle_avoid_zone( 

328 current_state, 

329 FIELD_DIMENTIONS.PENALTY_AREA_X_FRONT, 

330 FIELD_DIMENTIONS.X_LENGTH / 2.0, 

331 -half_width, 

332 half_width, 

333 ) 

334 ] 

335 if current_state.world_states["ball_position_relative"] == BALL_POSITION.UNKNOWN: 

336 return [] 

337 assert current_state.ball_position_local is not None 

338 return [ 

339 circle_avoid_zone( 

340 current_state.ball_position_local, SET_PLAY_KEEP_AWAY_DISTANCE 

341 ) 

342 ]

348def retrieve_team_data_from_gc_teams(gc_teams: list[TeamInfo]) -> TeamInfo | None: 

349 """ 

350 gcから来るデータのteamsから自チームのデータを取り出す 

351 自チームが含まれていない場合はNoneを返す。 

352 (simでは味方チームと敵チームがそれぞれ自チーム向けのGCメッセージを同じtopicへ 

353 流すため、先頭のチームへフォールバックすると相手チームのスコアやペナルティを 

354 自分のものとして読んでしまう) 

355 """ 

356 our_team_id = DEFAULT_STRATEGY_CONFIG["common_config"]["team_id"] 

357 for team in gc_teams: 

358 if team.team_number == our_team_id: 

359 return team 

360 return None
