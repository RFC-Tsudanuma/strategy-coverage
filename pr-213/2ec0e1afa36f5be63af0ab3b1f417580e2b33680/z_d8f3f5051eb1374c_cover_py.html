39def is_ball_in_danger_zone(s: StrategyState) -> bool: 

40 """ 

41 ボールが危険ゾーンにあるかどうかを判定する関数 

42 危険ゾーンにあればディフェンダーは何らかの対応の必要がある 

43 """ 

44 # 情報が無い場合はFalse 

45 if s.world_states["ball_position_relative"] == BALL_POSITION.UNKNOWN: 

46 return False 

47 

48 # 自陣ゴール前300cm以内を危険ゾーンとする（LEFT-view視点での閾値） 

49 DANGER_ZONE_X_LEFT_VIEW = -FIELD_DIMENTIONS.X_LENGTH / 2 + 300.0 

50 # 中央直径400cmを危険ゾーンとする 

51 DANGER_ZONE_Y = 200.0 

52 ball_gl = utils.geometry.convert_local_to_global( 

53 s.self_position_global, s.ball_position_local 

54 ) 

55 ball_x_left = utils.geometry.to_left_view_x(s, ball_gl.x) 

56 if ball_x_left < DANGER_ZONE_X_LEFT_VIEW and abs(ball_gl.y) < DANGER_ZONE_Y: 

57 return True 

58 return False

61def vector_robot_to_ball_directs_goal(s: StrategyState) -> bool: 

62 """ 

63 ロボットからボールへのベクトルがゴール方向かどうかを判定する関数 

64 get_goalpost_localは既にOUR_FIELDを考慮した相手ゴールのローカル座標を返すため、 

65 ロボット原点(0,0)からポストへのベクトルはpos_a, pos_b自身に等しい。 

66 """ 

67 if s.world_states["ball_position_relative"] == BALL_POSITION.UNKNOWN: 

68 return False 

69 # ボールへのベクトル（ローカル: ロボット原点が0,0） 

70 vec_ball = np.array([s.ball_position_local.x, s.ball_position_local.y]) 

71 # ロボットからゴールポストへのベクトル（ローカル） 

72 pos_a, pos_b = ( 

73 utils.geometry.get_goalpost_local(s)[0], 

74 utils.geometry.get_goalpost_local(s)[1], 

75 ) 

76 post_a_vec = np.array([pos_a.x, pos_a.y]) 

77 post_b_vec = np.array([pos_b.x, pos_b.y]) 

78 # ボールのベクトルが上の2つのベクトルの間にあるならゴール方向を向かっている 

79 cross_a = np.cross(post_a_vec, vec_ball) 

80 cross_b = np.cross(vec_ball, post_b_vec) 

81 if (cross_a * cross_b >= 0) and ((cross_a >= 0) == (cross_b >= 0)): 

82 return True 

83 return False

86@contract.htn_method(task_name="cover_task", preconditions=[("role", ROLE.COVER)]) 

87def cover_task_method( 

88 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

89) -> list[tuple[str, WalkCommandPublisher, DecisionLogger]] | None: 

90 """ 

91 Define the actions for the cover role. 

92 """ 

93 # ペナライズから復帰して3秒間は指定位置に移動する 

94 if get_strategy_time() - s.last_unpenalized_time < 3.0: 

95 print_in_task_method( 

96 "Recently unpenalized. Walking to default position.", 

97 ) 

98 logger.task_executed(details="Recently unpenalized, moving to default position") 

99 return [ 

100 # 戻る時は首振りして周囲を見ながら帰る 

101 ("search_ball_by_neck_movement", pub, logger), 

102 ("walk_to_default_position", pub, logger), 

103 ] 

104 # ボールが近くにあって、ゴール内に向かう方向でなければタックルする 

105 if is_ball_in_danger_zone(s) and not vector_robot_to_ball_directs_goal(s): 

106 print_in_task_method("Ball in danger zone. Tackling.") 

107 logger.task_executed(details="Ball in danger zone, executing tackle_ball task") 

108 return [ 

109 # 蹴る時はボールを見ながら蹴る 

110 ("turn_neck_to_detected_ball", pub, logger), 

111 ("tackle_ball", pub, logger), 

112 ] 

113 # 特に問題ない場合は、移動して守りやすいポジションを取り続ける 

114 logger.task_executed(details="Approaching cover position") 

115 return [ 

116 # 取りあえずボールがある方向を向いておく 

117 ("turn_neck_to_detected_ball", pub, logger), 

118 ("approach_cover_position", pub, logger), 

119 ]
