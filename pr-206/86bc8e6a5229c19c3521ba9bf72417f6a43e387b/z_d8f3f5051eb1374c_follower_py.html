45# そもそもfollower_taskに切り替わる条件として情報共有で誰かapproacherでいることが前提 

46@contract.htn_method(task_name="follower_task") 

47def follower_task_method( 

48 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

49) -> list[tuple[str, WalkCommandPublisher, DecisionLogger]]: 

50 """ 

51 Define the actions for the follower role. 

52 """ 

53 # 味方ロボットが検出されない場合はapproacherに切り替わる 

54 tmp = get_shared_ally_positions(s) 

55 shared_allys = [x for x in tmp if x is not None] 

56 if not s.allies_position_local and len(shared_allys) == 0: 

57 print_in_real_action( 

58 "味方が見つかりませんでした。アタッカーの役割に切り替えます", 

59 ) 

60 logger.task_executed(details="No allies found, switching to approacher role") 

61 return [("approacher_task", pub, logger)] 

62 # ボールが見えており、味方もいる場合は追従アクションを実行 

63 logger.task_executed( 

64 details="Ball detected, executing approach_approacher_follower action" 

65 ) 

66 return [("approach_approacher_follower", pub, logger)] 

67 

68 

69# アクション 

70@contract.htn_action( 

71 preconditions=[ 

72 contract.StateNotCondition( 

73 key="ball_position_relative", value=BALL_POSITION.UNKNOWN 

74 ), 

75 ("allies_position_local", contract.BoolCondition.TRUE), 

76 ], 

77 effects=[], # effect特に無し TODO: 必要なら追加 

78) 

79def approach_approacher_follower( 

80 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

81) -> StrategyState | None: 

82 # 味方local、ボールlocal、自分のglobalがある場合 

83 if ( 

84 s.allies_position_local 

85 and s.world_states["ball_position_relative"] != BALL_POSITION.UNKNOWN 

86 ): 

87 # s.allies_position_localはPositionオブジェクトのリストなので、最初の要素を取得する 

88 first_ally_local_position = s.allies_position_local[0] 

89 

90 # 自分とボールの差分、味方とボールの差分を比較して 

91 # ボールのグローバル座標 

92 ball_gl = convert_local_to_global(s.self_position_global, s.ball_position_local) 

93 allies_gl = convert_local_to_global( 

94 s.self_position_global, first_ally_local_position 

95 ) 

96 

97 diff_ball_alies = calculate_distance_global(ball_gl, allies_gl) 

98 diff_ball_self_position = calculate_distance_global( 

99 ball_gl, s.self_position_global 

100 ) 

101 

102 if diff_ball_self_position > diff_ball_alies: 

103 # 味方と一定の距離を保って、追従する 

104 # ペナルティエリアよりボールが後ろにあるとそれ以上下がらない 

105 # LEFT-view視点で計算してconvert_hardcoded_pos_to_our_fieldで一度だけflipする 

106 ball_x_left = to_left_view_x(s, ball_gl.x) 

107 if ball_x_left > -450.0: 

108 follower_x_left = ball_x_left - 250.0 

109 else: 

110 follower_x_left = -450.0 

111 

112 # ボールのy座標から離れる（yは左右対称なので変換不要） 

113 if ball_gl.y > 0.0: 

114 allies_follower_pos_y = ball_gl.y - 250.0 

115 else: 

116 allies_follower_pos_y = ball_gl.y + 250.0 

117 

118 follower_position_global = convert_hardcoded_pos_to_our_field( 

119 s, 

120 Position( 

121 Position.COORD_GLOBAL, 

122 follower_x_left, 

123 allies_follower_pos_y, 

124 0.0, 

125 ), 

126 ) 

127 

128 # 目標位置とロボットの現在の位置のグローバル座標間の距離を計算 

129 distance_to_target = calculate_distance_global( 

130 s.self_position_global, follower_position_global 

131 ) 

132 

133 # ロボットが目標位置に十分に近づいたら、ボールの方向を向く 

134 if distance_to_target < 150.0: # 距離が150mm以内なら到着したとみなす 

135 # ローカル座標のボールの角度を計算 

136 _ball_angle = calculate_angle(s.ball_position_local) 

137 

138 # 首の動きと体の旋回を調整 

139 # 首をボールの方向へ 

140 # pub.move_neck_yaw(yaw_angle=-ball_angle) 

141 # 体の向きを調整するための旋回コマンドを生成 

142 cmd = action_functions.generate_turn_command(s.ball_position_local) 

143 pub.send_walk_command(cmd) 

144 print_in_real_action("Turning to face the ball") 

145 logger.action_executed(details="Turning to face the ball") 

146 

147 return s 

148 

149 else: 

150 cmd = action_functions.move_to_target_global_pose( 

151 s, pub, follower_position_global 

152 ) 

153 if cmd is not None: 

154 pub.send_walk_command(cmd) 

155 print_in_real_action("Moving to follower position") 

156 logger.action_executed(details="Moving to follower position") 

157 return s 

158 else: 

159 # 自分の方がボールに近い場合、followerからapproacherに切り替えるなど、 

160 # 新しいタスクへ遷移するロジックをここに追加することを検討 

161 print_in_real_action( 

162 "Self is closer to the ball. Transitioning to a new task.", 

163 ) 

164 logger.action_postcondition_already_satisfied( 

165 details="Self is closer to ball than ally, no action taken" 

166 ) 

167 return s # シンプルに現在の状態を維持 

168 

169 # ボールが見えない場合 

170 else: 

171 print_in_real_action( 

172 "Ball is not visible. Moving to a default cover position.", 

173 ) 

174 

175 # ボールが見えない場合の自陣でボールが探しやすい場所に移動 

176 default_follower_position = convert_hardcoded_pos_to_our_field( 

177 s, 

178 Position( 

179 Position.COORD_GLOBAL, 

180 -300.0, 

181 0.0, 

182 0.0, 

183 ), 

184 ) 

185 cmd = action_functions.move_to_target_global_pose( 

186 s, pub, default_follower_position 

187 ) 

188 if cmd is not None: 

189 pub.send_walk_command(cmd) 

190 logger.action_executed( 

191 details="Moving to default cover position as ball is not visible" 

192 ) 

193 return s
