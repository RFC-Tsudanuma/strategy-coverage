206 def subscribe_function(self) -> None: 

207 print_in_strategy("subscribe_function thread started") 

208 try: 

209 self.ros_executor = MultiThreadedExecutor() 

210 self.ros_executor.add_node(self.subscriber) 

211 self.ros_executor.add_node(self.publisher) 

212 # Use spin_once with timeout instead of blocking spin 

213 while rclpy.ok() and not self.finish: 

214 self.ros_executor.spin_once(timeout_sec=0.1) 

215 print_in_strategy("[DEBUG] subscribe_function loop exited normally") 

216 except KeyboardInterrupt: 

217 print_in_strategy("Subscriber thread has been interrupted by keyboard.") 

218 self.finish = True 

219 except rclpy.executors.ExternalShutdownException: 

220 print_in_strategy("Subscriber thread has been shutdown externally.") 

221 self.finish = True 

222 except Exception as e: 

223 print_in_strategy(f"Subscriber thread error: {e}") 

224 self.finish = True 

225 finally: 

226 self.finish = True 

227 print_in_strategy("[DEBUG] subscribe_function finally block") 

228 if self.ros_executor: 

229 self.ros_executor.shutdown()

254 def find_plan(self, state: StrategyState) -> list | None: 

255 if self.finish: 

256 raise RuntimeError("Strategy has been finished.") 

257 # plan変数をtryブロックの外で初期化 

258 plan = None 

259 

260 try: 

261 plan = gtpyhop.find_plan(state, self.default_todo_list) 

262 if plan: 

263 print_in_strategy( 

264 f"[ Strategy ] Plan found for robot {self.robot_id}: " 

265 ) 

266 for i, action in enumerate(plan): 

267 print_in_strategy(f"---> {i}: {get_real_action_name(action)}") 

268 else: 

269 print_in_strategy( 

270 f"[ Strategy ] No plan found for robot {self.robot_id}" 

271 ) 

272 return None 

273 

274 except RoleChangeException as e: 

275 print_in_strategy(f"Role change requested: {e.new_role}") 

276 self._apply_role_change(e.new_role) 

277 # ロール変更後、新しいプランを再生成し、planに再代入する 

278 plan = gtpyhop.find_plan(self.current_state, self.default_todo_list) 

279 if not plan: # 再度プランが見つからない場合も考慮 

280 return None 

281 print_in_strategy( 

282 f"[ Strategy ] Plan found after role change for robot {self.robot_id}: " 

283 ) 

284 for i, action in enumerate(plan): 

285 print_in_strategy(f"---> {i}: {get_real_action_name(action)}") 

286 except Exception as e: 

287 self.finish = True 

288 print_in_strategy(f" An error occurred: {e}") 

289 import traceback 

290 

291 traceback.print_exc() 

292 self.logger.print_decision_log_for_debug() # エラー発生時に意思決定ログを出力 

293 raise RuntimeError("Strategy will be finished due to an error.") from e 

294 

295 return plan

297 def execute_action(self, action: str) -> typing.Any: 

298 # アクションを実行するロジックを実装 

299 # 成功した場合はTrue、失敗した場合はFalseを返す 

300 # 今はvirtual_actionが消えたので、恐らく利用しない 

301 from real_action.actions import REAL_ACTIONS_DICT 

302 

303 real_action = REAL_ACTIONS_DICT[get_real_action_name(action)] 

304 print_in_strategy(f"Executing action: {real_action.__name__}") 

305 try: 

306 args = action[1:] 

307 if len(args) > 0: 

308 result = real_action( 

309 self.current_state, *args 

310 ) # 引数ありの実アクションを実行 

311 else: 

312 result = real_action( 

313 self.current_state, 

314 self.publisher, 

315 self.logger, 

316 ) # 引数なしの実アクションを実行 

317 except RoleChangeException as e: 

318 print_in_strategy(f"Role change requested: {e.new_role}") 

319 self._apply_role_change(e.new_role) 

320 result = ACTION_RESULT.SUCCESS 

321 return result

323 def finish_strategy(self) -> None: 

324 self.finish = True 

325 # 戦略の終了処理を実装 

326 print_in_strategy(f"Strategy finished for robot {self.robot_id}") 

327 

328 # まずゲームコントローラーとの通信を終了 

329 self.brain_comm.clearup_game_controller_broadcast() 

330 

331 # rclpyのシャットダウンを先に実行（これによりexecutorのspin()が停止する） 

332 print_in_strategy("Shutting down rclpy...") 

333 try: 

334 rclpy.shutdown() 

335 except Exception as e: 

336 print_in_strategy(f"rclpy.shutdown() failed: {e}") 

337 

338 # スレッドの終了を待つ（タイムアウトを設定） 

339 if self.sub_thread and self.sub_thread.is_alive(): 

340 print_in_strategy("Waiting for subscriber thread to finish...") 

341 self.sub_thread.join(timeout=3.0) 

342 if self.sub_thread.is_alive(): 

343 print_in_strategy("Warning: Subscriber thread did not finish in time") 

344 

345 # ノードを破棄（既に破棄されている可能性があるのでtry-catchで囲む） 

346 if self.subscriber: 

347 self.subscriber.destroy_node() 

348 

349 if self.publisher: 

350 self.publisher.destroy_node()

545 def update_strategy_hz(self) -> float: 

546 # 周期の計算 

547 now = get_strategy_time() 

548 hz = 1.0 / (now - self.prev_strategy_time) 

549 self.current_state.last_strategy_process_time = now - self.prev_strategy_time 

550 print_in_strategy(f"[Hz info] {hz}hz") 

551 self.prev_strategy_time = now 

552 return hz
