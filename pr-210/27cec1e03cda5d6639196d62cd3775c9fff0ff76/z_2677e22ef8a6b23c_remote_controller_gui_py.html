313 def _apply_preset(self, fields: tuple[str, ...]) -> None: 

314 """プリセットで指定された bool フィールドだけを選択状態にする.""" 

315 for name, btn in self.button_widgets.items(): 

316 btn.setChecked(name in fields) 

317 self.status_label.setText(f"Preset applied: {', '.join(fields)}") 

318 

319 def _on_clear_clicked(self) -> None: 

320 for btn in self.button_widgets.values(): 

321 btn.setChecked(False) 

322 self.lx_spin.setValue(0.0) 

323 self.ly_spin.setValue(0.0) 

324 self.rx_spin.setValue(0.0) 

325 self.ry_spin.setValue(0.0) 

326 self.event_spin.setValue(0) 

327 self.hat_pos_spin.setValue(0) 

328 self.status_label.setText("Cleared all selections.") 

329 

330 def _on_send_clicked(self) -> None: 

331 msg = self._build_message() 

332 self.publisher.publish(msg) 

333 active = [name for name in BOOL_FIELDS if getattr(msg, name)] 

334 timestamp = datetime.now().strftime("%H:%M:%S") 

335 if active: 

336 self.status_label.setText( 

337 f"[{timestamp}] Sent: buttons={'+'.join(active)}" 

338 ) 

339 else: 

340 self.status_label.setText(f"[{timestamp}] Sent (no buttons pressed).") 

341 self.get_logger().info(f"Published RemoteControllerState: buttons={active}") 

342 

343 def _build_message(self) -> RemoteControllerState: 

344 msg = RemoteControllerState() 

345 for name, btn in self.button_widgets.items(): 

346 setattr(msg, name, btn.isChecked()) 

347 msg.lx = float(self.lx_spin.value()) 

348 msg.ly = float(self.ly_spin.value()) 

349 msg.rx = float(self.rx_spin.value()) 

350 msg.ry = float(self.ry_spin.value()) 

351 msg.event = int(self.event_spin.value()) 

352 msg.hat_pos = int(self.hat_pos_spin.value()) 

353 return msg
