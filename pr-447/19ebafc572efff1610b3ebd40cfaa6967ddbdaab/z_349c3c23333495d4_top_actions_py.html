307def course_cut_pass_lane( 

308 s: BallKnownState, pub: WalkCommandPublisher, logger: DecisionLogger 

309) -> BallKnownState | None: 

310 """ 

311 敵のセットプレー中の approacher 用。ボールから keep-away 距離を保ったまま、 

312 1 本のパスコース (ボール → 最も自ゴールに近い受け手) の線上に立つ。 

313 受け手が居なければシュートコース (ボール → 自ゴール中央) を切る。 

314 共有マップに敵が居るときだけ呼ぶこと。 

315 """ 

316 ball_global = _course_cut_ball_global(s) 

317 # ponytail: 受け手は「最も自ゴールに近い敵」で選ぶ。脅威評価 (docs §4) で選ぶのは効果が見えてから 

318 # 2 番目の受け手は top が切る (set_play_top_receiver) 

319 receivers = set_play_lane_receivers(s, ball_global) 

320 aim = receivers[0] if receivers else get_ally_goal_center_pos_global(s) 

321 direction = math.atan2(aim.y - ball_global.y, aim.x - ball_global.x) 

322 keep_away = setting.SET_PLAY_KEEP_AWAY_DISTANCE + _course_cut_arrive_distance() 

323 lane_point = Position( 

324 Position.COORD_GLOBAL, 

325 ball_global.x + keep_away * math.cos(direction), 

326 ball_global.y + keep_away * math.sin(direction), 

327 ) 

328 lane_point = _out_of_their_penalty_area(s, ball_global, lane_point) 

329 # 到着判定 (arrive_distance) に入った所で止まると、来た側へその分コースから横にずれたまま 

330 # 蹴られる (sim で 0.2 m ほど)。自分から見てコース上の点の arrive_distance 先を目指し、 

331 # ちょうどコース上の点で到着になるようにする。 

332 me = s.self_position_global 

333 to_lane = calculate_distance_global(me, lane_point) 

334 if to_lane > 1.0: 

335 overshoot = _course_cut_arrive_distance() / to_lane 

336 target_global = Position( 

337 Position.COORD_GLOBAL, 

338 lane_point.x + (lane_point.x - me.x) * overshoot, 

339 lane_point.y + (lane_point.y - me.y) * overshoot, 

340 ) 

341 else: 

342 target_global = lane_point 

343 # 外から近づくとオーバーシュートでエリアに入り直すので、もう一度押し出す 

344 target_global = _out_of_their_penalty_area(s, ball_global, target_global) 

345 cmd = _course_cut_walk_command(s, pub, target_global, s.ball_position_local) 

346 if cmd is None: 

347 logger.action_postcondition_already_satisfied( 

348 details="At pass lane position (or no valid path)" 

349 ) 

350 return s 

351 pub.send_walk_command(cmd) 

352 logger.action_executed(details="Moving onto the pass lane") 

353 return None

373def _course_cut_assignment_target(s: BallKnownState) -> Position | None: 

374 """コース切り守備で自分に割り当てられた目標 (割り当てが無ければ None)。 

375 敵のセットプレー中は keep-away 距離の外へ押し出す。""" 

376 ball_global = _course_cut_ball_global(s) 

377 plan = plan_course_cut_defence_for_self( 

378 s, ball_global, load_course_cut_defence_config() 

379 ) 

380 assignment = plan.for_robot(s.robot_id) 

381 if assignment is None: 

382 return None 

383 target_global = assignment.target 

384 if s.world_states["set_play"] in SET_PLAY_STATES: 

385 # 敵のセットプレー中はボールから離れている義務があるので、同じコース上で 

386 # ボールから遠ざかる向きに押し出す。目標の手前 (arrive_distance) で止まっても 

387 # 離れているように余裕を足す。 

388 keep_away = setting.SET_PLAY_KEEP_AWAY_DISTANCE + _course_cut_arrive_distance() 

389 if calculate_distance_global(ball_global, target_global) < keep_away: 

390 direction = math.atan2( 

391 target_global.y - ball_global.y, target_global.x - ball_global.x 

392 ) 

393 target_global = Position( 

394 Position.COORD_GLOBAL, 

395 ball_global.x + keep_away * math.cos(direction), 

396 ball_global.y + keep_away * math.sin(direction), 

397 ) 

398 target_global = _out_of_their_penalty_area(s, ball_global, target_global) 

399 return target_global

402def set_play_top_receiver(s: BallKnownState) -> Position | None: 

403 """ 

404 敵のセットプレーで top が切る受け手。approacher が共有マップの先頭の受け手を切るので、top は 

405 共有マップの 2 番目の受け手を切る。 

406 共有マップは 2 s ほどで敵を忘れ、そのたびに受け手が変わると見張りの向きごと崩れるので、 

407 共有マップに受け手が 1 体しかなければ自分のカメラの記憶 (enemies_seen_by_self) から、 

408 approacher の受け手から最も遠い敵を選ぶ。記憶には同じ敵の数秒前の位置が merge 距離より離れて 

409 残ることがあり、それは approacher の受け手の近くにあるので「最も遠い」で除ける。 

410 記憶にもう 1 体が無ければ (まだ 1 体しか見つけていない)、見つけた敵を切りに行く。 

411 ponytail: 「最も遠い」は受け手 2 体 (3v3) の前提。3 体以上なら記憶にも脅威順が要る 

412 敵のセットプレーでない、または受け手が分からなければ None。 

413 """ 

414 if s.world_states["set_play"] not in SET_PLAY_STATES: 

415 return None 

416 receivers = set_play_lane_receivers(s, _course_cut_ball_global(s)) 

417 if len(receivers) >= 2: 

418 return receivers[1] 

419 seen = [ 

420 e 

421 for e in enemies_seen_by_self_except_kicker(s, get_strategy_time()) 

422 if not in_their_goal_area(s, e) 

423 ] 

424 if not receivers: 

425 return min( 

426 seen, 

427 key=lambda e: calculate_distance_global(s.self_position_global, e), 

428 default=None, 

429 ) 

430 approachers_receiver = receivers[0] 

431 merge_distance = setting.STRATEGY_SHARED_CONFIG["course_cut_cover"][ 

432 "enemies_seen_merge_distance_cm" 

433 ] 

434 others = [ 

435 e 

436 for e in seen 

437 if calculate_distance_global(e, approachers_receiver) > merge_distance 

438 ] 

439 if not others: 

440 return approachers_receiver if seen else None 

441 return max(others, key=lambda e: calculate_distance_global(approachers_receiver, e))

444def set_play_other_enemy_known(s: BallKnownState, receiver: Position) -> bool: 

445 """ 

446 top がマークする受け手 (receiver) 以外の受け手を、共有マップか自分のカメラの記憶で知っているか。 

447 ponytail: 記憶に残った受け手自身の数秒前の位置 (merge 距離より離れたもの) も「もう 1 体」に数える。 

448 誤って探索をやめることが問題になったら、記憶の追跡を直す 

449 """ 

450 merge_distance = setting.STRATEGY_SHARED_CONFIG["course_cut_cover"][ 

451 "enemies_seen_merge_distance_cm" 

452 ] 

453 known = [ 

454 *set_play_lane_receivers(s, _course_cut_ball_global(s)), 

455 *enemies_seen_by_self_except_kicker(s, get_strategy_time()), 

456 ] 

457 return any(calculate_distance_global(e, receiver) > merge_distance for e in known)

460def _set_play_mark_target(s: BallKnownState, receiver: Position) -> Position: 

461 """ボール → receiver の延長上で、受け手の背後 set_play_mark_distance の位置 (ボールと自分で受け手を挟む)。 

462 フィールド内と敵ペナルティエリアの外に収める。""" 

463 ball_global = _course_cut_ball_global(s) 

464 direction = math.atan2(receiver.y - ball_global.y, receiver.x - ball_global.x) 

465 mark = setting.STRATEGY_SHARED_CONFIG["course_cut_cover"][ 

466 "set_play_mark_distance_cm" 

467 ] 

468 half_x = setting.FIELD_DIMENTIONS.X_LENGTH / 2.0 

469 half_y = setting.FIELD_DIMENTIONS.Y_LENGTH / 2.0 

470 target_global = Position( 

471 Position.COORD_GLOBAL, 

472 min(max(receiver.x + mark * math.cos(direction), -half_x), half_x), 

473 min(max(receiver.y + mark * math.sin(direction), -half_y), half_y), 

474 ) 

475 return _out_of_their_penalty_area(s, ball_global, target_global)

478def _out_of_their_penalty_area( 

479 s: BallKnownState, ball_global: Position, target_global: Position 

480) -> Position: 

481 """ 

482 敵ペナルティエリア内からの敵セットプレー (ゴールキックは必ずこれ) では、守備側は 

483 ボールが in play になるまでペナルティエリアの外に居る義務がある (HSL rules §13.3.2, §16.2)。 

484 目標がエリア内なら、ボール → 目標の延長上でエリアを出た所まで押し出す。 

485 目標の手前 (arrive_distance) で止まってもエリアの外に居るように、その分エリアを広げて扱う。 

486 """ 

487 our_field = typing.cast(OUR_FIELD, s.world_states["our_field"]) 

488 margin = _course_cut_arrive_distance() 

489 front_x = setting.FIELD_DIMENTIONS.PENALTY_AREA_X_FRONT - margin 

490 half_width = setting.FIELD_DIMENTIONS.PENALTY_AREA_WIDTH / 2.0 + margin 

491 

492 def in_their_area(p: Position) -> bool: 

493 return ( 

494 abs(p.x) > front_x 

495 and abs(p.y) < half_width 

496 and not is_object_in_our_field(our_field, p) 

497 ) 

498 

499 if not (in_their_area(ball_global) and in_their_area(target_global)): 

500 return target_global 

501 dx = target_global.x - ball_global.x 

502 dy = target_global.y - ball_global.y 

503 goal_side = math.copysign(1.0, ball_global.x) 

504 # ボールから目標方向へ進んで、前端 (x) か横端 (y) を最初に越える所 

505 exits = [] 

506 if dx * goal_side < 0: 

507 exits.append((goal_side * front_x - ball_global.x) / dx) 

508 if dy != 0: 

509 exits.append((math.copysign(half_width, dy) - ball_global.y) / dy) 

510 if not exits: 

511 # 真っ直ぐゴールラインへ向かうコースは外へ出ないので、前端へ下がる 

512 return Position(Position.COORD_GLOBAL, goal_side * front_x, target_global.y) 

513 t = min(exits) 

514 return Position( 

515 Position.COORD_GLOBAL, ball_global.x + dx * t, ball_global.y + dy * t 

516 )

521 pub: WalkCommandPublisher, 

522 target_global: Position, 

523 watch_local: Position | None, 

524) -> WalkCommand | None: 

525 """ 

526 target_global へ、watch_local を視野に入れたまま横歩き・後ろ歩きで向かう歩行コマンド。 

527 到着済み、または有効な経路が無ければ None。 

528 """ 

529 pub.send_target_position(target_global) 

530 if ( 

531 calculate_distance_global(s.self_position_global, target_global) 

532 <= _course_cut_arrive_distance() 

533 ): 

534 return None 

535 

536 # 目標へ回り込む途中でボールを蹴らないよう、ボールも避けて通る 

537 obstacles = [*get_all_obstacles(s), s.ball_position_local] 

538 planned_path, _ = pathplan.find_path_to_target( 

539 convert_global_to_local(s.self_position_global, target_global), 

540 obstacles, 

541 clearance=setting.DEFAULT_STRATEGY_CONFIG["common_config"][ 

542 "pathplan_clearance" 

543 ], 

544 clearance_factor=pathplan.distance_clearance_factor( 

545 s.ball_position_local, 

546 obstacles, 

547 clearance=setting.DEFAULT_STRATEGY_CONFIG["common_config"][ 

548 "pathplan_clearance" 

549 ], 

550 ), 

551 publisher=pub, 

552 ) 

553 if not action_functions.validate_path(planned_path): 

554 return None 

555 # 見ている対象から目を離さないよう、cover と同じく見ながら到達モードで動く
