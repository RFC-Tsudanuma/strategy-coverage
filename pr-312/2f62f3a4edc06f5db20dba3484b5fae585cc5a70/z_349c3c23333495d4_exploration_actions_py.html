52# do_flontier_exploration のモード判定で使う閾値。 

53# 直近に観測したボールが自分から NEAR_FIRST_RADIUS_CM 以内 かつ 

54# 観測してから LAST_VISIBLE_BALL_STALE_SEC 以内なら NEAR_FIRST モードを選ぶ。 

55NEAR_FIRST_RADIUS_CM = 150.0 

56LAST_VISIBLE_BALL_STALE_SEC = 10.0 

57 

58 

59@contract.htn_action() 

60def do_flontier_exploration( 

61 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

62) -> StrategyState | None: 

63 """ 

64 フロンティア探索を行うアクション。 

65 last_visible_ball が新しく自分から NEAR_FIRST_RADIUS_CM 以内なら NEAR_FIRST モード、 

66 それ以外（遠い／観測なし／古すぎ）は FAR_FIRST モードでフロンティア探索を開始する。 

67 フロンティア探索を停止するのは、アクションを使った実装だと煩雑になるので、使わずに停止させても良い。 

68 """ 

69 print_in_real_action("Executing do_flontier_exploration().") 

70 if not action_functions.need_flontier_exploration(s): 

71 logger.action_precondition_not_satisfied( 

72 "No need to perform flontier_exploration" 

73 ) 

74 return None 

75 

76 last_ball, last_ts = s.last_visible_ball 

77 is_recent = ( 

78 last_ball is not None 

79 and (get_strategy_time() - last_ts) <= LAST_VISIBLE_BALL_STALE_SEC 

80 ) 

81 if ( 

82 is_recent 

83 and calculate_distance_global(s.self_position_global, last_ball) 

84 <= NEAR_FIRST_RADIUS_CM 

85 ): 

86 mode = DoFlontierExploration.Request.MODE_EXPLORE_NEAR_FIRST 

87 mode_label = "NEAR_FIRST" 

88 else: 

89 mode = DoFlontierExploration.Request.MODE_EXPLORE_FAR_FIRST 

90 mode_label = "FAR_FIRST" 

91 

92 # 複数ロボットでの役割分担: 奇数 ID は時計回り、偶数 ID は反時計回り。 

93 # FILL_FEET 時は flontier_exp 側で NONE 固定にしているのでここでは方向だけを渡す。 

94 if s.robot_id % 2 == 1: 

95 rotation_direction = DoFlontierExploration.Request.ROTATION_CW 

96 rotation_label = "CW" 

97 else: 

98 rotation_direction = DoFlontierExploration.Request.ROTATION_CCW 

99 rotation_label = "CCW" 

100 

101 pub.do_flontier_exploration( 

102 robot_id=s.robot_id, mode=mode, rotation_direction=rotation_direction 

103 ) 

104 logger.action_executed( 

105 details=f"Performing frontier exploration (mode={mode_label}, rot={rotation_label})" 

106 ) 

107 return s

110@contract.htn_action( 

111 preconditions=[ 

112 ("need_flontier_exploration", contract.BoolCondition.FALSE), 

113 ] 

114) 

115def stop_flontier_exploration( 

116 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

117) -> StrategyState | None: 

118 """ 

119 フロンティア探索を停止するアクション 

120 """ 

121 if not action_functions.need_flontier_exploration(s): 

122 print_in_real_action("Executing stop_flontier_exploration().") 

123 # フロンティア探索モードを停止 

124 pub.stop_flontier_exploration(robot_id=s.robot_id) 

125 logger.action_executed(details="Stopped frontier exploration") 

126 return s 

127 else: 

128 print_in_real_action("No need to stop flontier_exploration.") 

129 logger.action_precondition_not_satisfied("No need to stop flontier_exploration") 

130 return None

133@contract.htn_action( 

134 preconditions=[ 

135 ("need_flontier_exploration", contract.BoolCondition.TRUE), 

136 ("role", ROLE.KEEPER),

138) 

139def keeper_explore_back_to_position( 

140 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

141) -> StrategyState | None: 

142 keeper_center_pos = Position( 

143 coord=Position.COORD_GLOBAL, 

144 x=-setting.FIELD_DIMENTIONS.X_LENGTH / 2, 

145 y=0.0, 

146 theta=0.0, 

147 ) 

148 if not are_target_poses_near( 

149 s.self_position_global, keeper_center_pos, 30, 30, math.radians(15.0) 

150 ): 

151 cmd = action_functions.move_to_target_global_pose(s, pub, keeper_center_pos) 

152 if cmd is not None: 

153 pub.send_walk_command(cmd) 

154 logger.action_executed(details="Moving to keeper position") 

155 return s 

156 else: 

157 logger.action_postcondition_already_satisfied( 

158 details="Already moved to keepre position" 

159 ) 

160 return s
