1"""debug モードの policy kick (task_policy_kick_planned) で、蹴る向きと回り込みの要否を決める。 

2 

3共有設定 kick_value_config: 

4 enable: true なら計測したキック方向の誤差の分布 (kick_value_planner) で向きを選ぶ。 

5 mode: shot はゴールを狙い、誤差の中央値を見込んで真ん中へ蹴る。mode: plan は 

6 価値反復で向きと回り込みを選び、解けるまでは下の従来の決め方に戻る。 

7 foot_selection: both なら蹴り足も状況ごとに選ぶ (fixed: 設定 kick_foot の足)。 

8 align_max_turn_deg: > 0 なら、蹴る向きとロボットの向きの差がこれを超えるときに、 

9 先にボールの周りを回って向きを合わせる。policy は蹴りながら大きく回ると 

10 向きが外れやすい (sim、実機相当の遅延: 旋回 135° 超で実行誤差の中央値 -16°)。 

11""" 

12

15import math 

16from dataclasses import dataclass 

17from typing import Any

19import kick_value_planner 

20import setting 

21from constants import KICK_FOOT, OUR_FIELD 

22from field_dimensions import load_field_dimensions_m 

23from state import BallKnownState 

24from utils.geometry import ( 

25 calculate_kick_direction_world,

32@dataclass(frozen=True) 

33class KickDecision: 

34 direction: float # 蹴る向き (ワールド座標の角度) [rad] 

35 align: bool # True: 蹴る前にボールの周りを回って向きを合わせる 

36 source: str # "goal_shot" / "value_iteration" / "goal_center" 

37 expected_time: float = math.nan # 価値反復の得点までの時間の期待値 [s] 

38 # このキックが両ポストの間へ向かう確率 (shot) / 得点確率 (価値反復) 

39 p_goal: float = math.nan 

40 foot: KICK_FOOT | None = None # None なら設定 kick_foot 

41

43def kick_value_config() -> dict[str, Any]: 

44 return dict(setting.STRATEGY_SHARED_CONFIG.get("kick_value_config", {})) 

45 

46 

47def needs_align(direction: float, heading: float, cfg: dict[str, Any]) -> bool: 

48 limit = float(cfg.get("align_max_turn_deg", 0.0)) 

49 return limit > 0.0 and abs(normalize_angle(direction - heading)) > math.radians( 

50 limit 

51 )

54def planner() -> kick_value_planner.Planner | None: 

55 """有効ならプランナ (mode: plan は初回の呼び出しで価値表を解き始める)。""" 

56 return kick_value_planner.get_planner( 

57 kick_value_config(), 

58 setting.STRATEGY_CONFIG_DIRECTORY, 

59 load_field_dimensions_m(), 

60 kick_foot=setting.STRATEGY_SHARED_CONFIG.get("kick_foot", "left"), 

61 ) 

62 

63 

64def decide_kick(s: BallKnownState) -> KickDecision: 

65 cfg = kick_value_config() 

66 heading = s.self_position_global.theta 

67 planner_ = planner() 

68 if planner_ is not None: 

69 ball = convert_local_to_global(s.self_position_global, s.ball_position_local) 

70 plan = planner_.plan( 

71 (ball.x / setting.CONVERT_SCALE, ball.y / setting.CONVERT_SCALE), 

72 heading, 

73 mirrored=s.world_states["our_field"] == OUR_FIELD.RIGHT, 

74 ) 

75 if plan is not None: 

76 return KickDecision( 

77 direction=plan.direction, 

78 align=plan.aligned or needs_align(plan.direction, heading, cfg), 

79 source=planner_.name, 

80 expected_time=plan.expected_time, 

81 p_goal=plan.p_goal, 

82 foot=plan.foot, 

83 ) 

84 direction = calculate_kick_direction_world(s, get_goal_center_pos(s)) 

85 return KickDecision( 

86 direction=direction, 

87 align=needs_align(direction, heading, cfg), 

88 source="goal_center", 

89 )
