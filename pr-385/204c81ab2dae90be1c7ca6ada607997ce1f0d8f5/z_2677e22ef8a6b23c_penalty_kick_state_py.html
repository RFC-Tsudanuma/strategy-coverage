25"""試合中(PK戦ではない)自チームPKの、tickをまたいで持ち越す状態。 

26 

27gtpyhop は探索のたびに state をコピーして使い捨てるため、tickをまたぐ値は 

28StrategyState (s.penalty_kick) 側に持たせ、更新は find_plan() の前後で 

29StrategyRunner が本クラスのメソッドを呼んで行う。PK用の状態と更新規則は 

30すべて本クラスに集め、StrategyRunner はいつ呼ぶかだけを知る。 

31""" 

32 

33import math 

34import random 

35from typing import TYPE_CHECKING 

36 

37import setting 

38from constants import GAME_STATE, ROBOT_MODE, SET_PLAY, SET_PLAY_BALL 

39 

40if TYPE_CHECKING: 

41 from futbol_msgs.msg import DecisionLog 

42 from state import StrategyState 

43 

44# 二度触り検知に使うキックアクション名。 

45# penalty_kick_role_task_select 経由のキッカーは必ず task_penalty_kick_random_aim_kick 

46# を通り、キックはこのアクション1つに一本化されている(名前がずれると誤検知/未検知になる)。 

47PENALTY_KICK_ACTION_NAME = "kick_motion_to_penalty_kick_random_post" 

48 

49 

50class PenaltyKickState: 

51 def __init__(self) -> None: 

52 # 自チームPKでキッカーが既に一度ボールに触れたか(二度触り防止)。 

53 self.ball_touched: bool = False 

54 # kick_motion が一度でも実行されたか(sticky)。実行された同tickに 

55 # ball_vel_prediction がまだ得られず cancel_kick_after_kick がキャンセル 

56 # することがあるため、ボール移動の確認を後続tickまで持ち越すのに使う。 

57 self.kick_attempted: bool = False 

58 # キッカーがどちらのポストを狙うか(True=goalpost_local[0]側)。 

59 # PLAYING開始時に一度だけ抽選し、蹴り終えるまで同じ側を狙い続ける。 

60 self.aim_prefers_post_a: bool | None = None 

61 

62 def reset(self) -> None: 

63 self.ball_touched = False 

64 self.kick_attempted = False 

65 self.aim_prefers_post_a = None 

66 

67 def update_before_plan(self, s: "StrategyState") -> None: 

68 """find_plan() の前に呼ぶ。キッカーが動く最初のtickから狙う側を確定させる。""" 

69 if self._is_our_penalty_kick_playing(s) and self.aim_prefers_post_a is None: 

70 self.aim_prefers_post_a = random.choice([True, False]) 

71 

72 def update_after_plan( 

73 self, s: "StrategyState", decision_log: "DecisionLog" 

74 ) -> None: 

75 """find_plan() の後・決定ログがクリアされる前に呼ぶ。二度触りを検知する。""" 

76 if not self._is_our_penalty_kick_playing(s): 

77 self.reset() 

78 return 

79 if s.robot_mode == ROBOT_MODE.STOP: 

80 # STOPモード(プラン確認用にコマンド送信を止めるデバッグモード)では 

81 # WalkCommandPublisher.stop_cmd_sending_mode によりキック自体は 

82 # 送信されないが、action_executedログは変わらず記録されるため、 

83 # ここで更新すると誤って二度触り扱いになる。STOP中は更新しない。 

84 return 

85 s.last_plan.update_log(decision_log) 

86 if s.last_plan.is_action_executed(PENALTY_KICK_ACTION_NAME): 

87 self.kick_attempted = True 

88 # 実行されたtickだけでなく後続tickも含めて、ボール移動を確認できるまで待つ。 

89 if self.kick_attempted and self._ball_confirmed_moving(s): 

90 self.ball_touched = True 

91 

92 @staticmethod 

93 def _is_our_penalty_kick_playing(s: "StrategyState") -> bool: 

94 return ( 

95 s.world_states["set_play"] == SET_PLAY.PENALTY_KICK 

96 and s.kickOffTeam == setting.get_team_id() 

97 and s.world_states["game_state"] == GAME_STATE.PLAYING 

98 ) 

99 

100 @staticmethod 

101 def _ball_confirmed_moving(s: "StrategyState") -> bool: 

102 # 速度予測が取れている間は速度閾値で見る(数tickで確定する速い経路)。 

103 # 弱いキックや速度予測が持続的に得られない場合でも、ペナルティマークからの 

104 # 位置移動(set_play_ball、ノイズ耐性のある判定)が閾値を超えれば同じく 

105 # 「動いた」と確定できる。どちらか一方が成立すればよい。 

106 if s.world_states["set_play_ball"] == SET_PLAY_BALL.MOVED: 

107 return True 

108 if s.ball_vel_prediction is None: 

109 return False 

110 ball_moved_vel_threshold: float = setting.DEFAULT_STRATEGY_CONFIG[ 

111 "common_config" 

112 ]["ball_moved_vel_threshold"] 

113 return bool( 

114 math.fabs(s.ball_vel_prediction.x_vel) > ball_moved_vel_threshold 

115 or math.fabs(s.ball_vel_prediction.y_vel) > ball_moved_vel_threshold 

116 )
