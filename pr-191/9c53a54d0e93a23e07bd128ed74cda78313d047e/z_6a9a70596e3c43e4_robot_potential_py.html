157 def gaussian_decay( 

158 center: int, 

159 i: int, 

160 j: int, 

161 max_val: int, 

162 sigma: float, 

163 limit_dist_sq: float, 

164 ) -> int: 

165 """ガウス分布に基づく減衰計算 

166 

167 inputs 

168 - center: 中心座標 

169 - i: x座標 

170 - j: y座標 

171 - max_val: ポテンシャルの最大値 

172 - sigma: ガウス分布の標準偏差 

173 - limit_dist_sq: 影響距離の二乗上限 

174 

175 returns 

176 - val: (i,j)のポテンシャルの値 

177 

178 -------------------------- 

179 """ 

180 dx = center - i 

181 dy = center - j 

182 dist_sq = float(dx * dx + dy * dy) 

183 if dist_sq > limit_dist_sq: 

184 return 0 

185 val = float(max_val) * np.exp(-dist_sq / (2.0 * sigma * sigma)) 

186 return round(val) 

187 

188 def sigma_and_limit_sq(max_val: int, decay_gain: float) -> tuple[float, float]: 

189 # 既存パラメータ(decay_gain)から、線形減衰の有効半径を近似してsigmaへ変換する 

190 safe_gain = max(float(decay_gain), 1e-6) 

191 approx_effect_radius = max(1.0, float(max_val) / safe_gain) 

192 sigma = max(1.0, approx_effect_radius / 3.0) 

193 return sigma, approx_effect_radius * approx_effect_radius
