65def _get_sqp_planner() -> "DiamondSQPPlanner": 

66 """stability_polytope_reduced.npz の安定領域を制約に持つ SQP プランナを返す（遅延生成）。""" 

67 global _sqp_planner 

68 if _sqp_planner is None: 

69 from diamond_sqp_planner import DiamondSQPPlanner 

70 

71 d = np.load(_STABILITY_POLYTOPE_NPZ) 

72 _sqp_planner = DiamondSQPPlanner( 

73 N=40, 

74 dt=0.1, 

75 Vx=float(d["Vx"]), 

76 Vy=float(d["Vy"]), 

77 W=float(d["W"]), 

78 poly_A=d["A"], # A @ u <= 1 の安定多面体（b は全て 1） 

79 ) 

80 return _sqp_planner

105def debug_go_to_target_pose( 

106 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

107) -> None: 

108 """ 

109 Action to go to a target pose in debug mode. 

110 DiamondSQPPlanner で安定領域（速度指令ダイヤモンド）を制約に速度指令を計画し、 

111 その先頭の制御 u0 = (vx, vy, wz)[ボディ座標系] を歩行コマンドとして送る。 

112 """ 

113 print_in_strategy("[Debug mode]: Going to target pose (SQP).") 

114 logger.action_executed(details="Goto target pose in DEBUG mode (SQP)") 

115 

116 target = pub.target_pose_from_user 

117 if target is None: 

118 pub.send_stop_command() 

119 return s 

120 

121 cur = s.self_position_global 

122 cfg = DEFAULT_STRATEGY_CONFIG["common_config"] 

123 

124 # 目標に十分近く、姿勢も合っていれば停止（既存の move_to_target_global_pose と同じ閾値）。 

125 dist_cm = math.hypot(target.x - cur.x, target.y - cur.y) # 位置は cm 

126 dtheta = abs( 

127 math.atan2( 

128 math.sin(target.theta - cur.theta), math.cos(target.theta - cur.theta) 

129 ) 

130 ) 

131 if ( 

132 dist_cm < cfg["move_pose_no_move_distance"] 

133 and dtheta < cfg["move_pose_no_move_angle"] 

134 ): 

135 pub.send_stop_command() 

136 return s 

137 

138 # SQP は m, m/s, rad で計画する。状態（self_position_global）は cm なので m に変換する。 

139 x_start = np.array([cur.x / 100.0, cur.y / 100.0, cur.theta]) 

140 goal = np.array([target.x / 100.0, target.y / 100.0, target.theta]) 

141 

142 global _sqp_warm_z, _sqp_warm_target 

143 target_key = (target.x, target.y, target.theta) 

144 z_init = _sqp_warm_z if target_key == _sqp_warm_target else None 

145 

146 planner = _get_sqp_planner() 

147 result = planner.plan(x_start, [], goal, z_init=z_init) 

148 _sqp_warm_z = result["z"] 

149 _sqp_warm_target = target_key 

150 

151 # 先頭の制御（ボディ座標系の速度指令）。m/s -> cm/s に戻して送る。wz は rad/s のまま。 

152 vx, vy, wz = result["controls"][0] 

153 pub.send_target_position(target) 

154 pub.send_walk_command(WalkCommand(vx * 100.0, vy * 100.0, wz)) 

155 return s

158@contract.htn_action(

162) 

163def debug_kick_ball_to_direction( 

164 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

165) -> None: 

166 """ 

167 Action to ram the ball in a user-specified direction (a "kick") in debug mode. 

168 target_pose_from_user.theta をキック方向とし、ボールを正確に通過しつつ接触時に 

169 ボディ前方 KICK_APPROACH_SPEED [m/s] で突き抜けるよう DiamondSQPPlanner で計画する。 

170 

171 ボール位置を終端にすると「ボール手前で減速・停止」してしまう（位置終端で止まる挙動）。 

172 そこで終端はキック方向にボールの先（総移動量 ~ v*T になる点）へ置き、ボール自身は 

173 経由点(waypoint)として正確に通過させる。これで停止せず一定速度で突っ込める。 

174 """ 

175 print_in_strategy("[Debug mode]: Kicking ball to direction (SQP).") 

176 logger.action_executed(details="Kick ball to direction in DEBUG mode (SQP)") 

177 

178 target = pub.target_pose_from_user 

179 ball_unknown = s.world_states["ball_position_relative"] == BALL_POSITION.UNKNOWN 

180 if target is None or ball_unknown: 

181 pub.send_stop_command() 

182 return s 

183 

184 cur = s.self_position_global 

185 # ボールのグローバル位置（cm）。キック方向はユーザ指定の向き（theta）。 

186 ball_global = convert_local_to_global(cur, s.ball_position_local) 

187 kick_theta = target.theta 

188 

189 planner = _get_sqp_planner() 

190 

191 # SQP は m, m/s, rad で計画する。位置（cm）は m に変換する。 

192 x_start = np.array([cur.x / 100.0, cur.y / 100.0, cur.theta]) 

193 ball_xy = np.array([ball_global.x / 100.0, ball_global.y / 100.0]) 

194 kick_dir = np.array([math.cos(kick_theta), math.sin(kick_theta)]) 

195 

196 # 終端をボールの先に置く。総移動量が v*T になるようフォロースルー距離を決めると、 

197 # 経路全体（=接触時）がほぼ KICK_APPROACH_SPEED の一定速度になる。 

198 horizon_t = planner.N * planner.dt 

199 proj = float( 

200 (ball_xy - x_start[:2]) @ kick_dir 

201 ) # ロボット->ボールのキック方向成分[m] 

202 follow = max(MIN_FOLLOW_THROUGH_M, KICK_APPROACH_SPEED * horizon_t - proj) 

203 goal_xy = ball_xy + kick_dir * follow 

204 goal = np.array([goal_xy[0], goal_xy[1], kick_theta]) 

205 # 接触時の終端速度：ヘディング方向（ボディ前方）へ KICK_APPROACH_SPEED で突っ込む。 

206 goal_vel = np.array([KICK_APPROACH_SPEED, 0.0, 0.0]) 

207 

208 # 経由点：ボール（必ず通過）。さらにロボットが十分手前にいれば、キック方向の手前に 

209 # アプローチ経由点を1つ足して早めにキック直線へ乗せる（直前の急旋回を避ける）。 

210 # 近い／通り過ぎている場合は後退を招くので足さない。 

211 waypoints = [tuple(ball_xy)] 

212 if proj > KICK_LINEUP_DIST_M + KICK_LINEUP_MARGIN_M: 

213 lineup_xy = ball_xy - kick_dir * KICK_LINEUP_DIST_M 

214 waypoints = [tuple(lineup_xy), tuple(ball_xy)] 

215 

216 global _sqp_kick_warm_z, _sqp_kick_warm_target 

217 target_key = (ball_global.x, ball_global.y, kick_theta) 

218 z_init = _sqp_kick_warm_z if target_key == _sqp_kick_warm_target else None 

219 

220 result = planner.plan(x_start, waypoints, goal, goal_vel=goal_vel, z_init=z_init) 

221 _sqp_kick_warm_z = result["z"] 

222 _sqp_kick_warm_target = target_key 

223 

224 # 先頭の制御（ボディ座標系の速度指令）。m/s -> cm/s に戻して送る。wz は rad/s のまま。 

225 vx, vy, wz = result["controls"][0] 

226 pub.send_target_position(ball_global) 

227 pub.send_walk_command(WalkCommand(vx * 100.0, vy * 100.0, wz)) 

228 return s
