1""" 

2TurnDriveController -- two-state near-unicycle controller. BOTH states solve a 

3small MPC (DiamondSQPPlanner) every tick and return the instantaneous velocity 

4command u0 = (vx, vy, wz). The objective is minimum time to the target pose: 

5the progress costs (w_run / w_run_theta) front-load the motion so the single 

6active channel saturates into a bang-bang-like accel / cruise / brake profile. 

7 

8 HEADING : wz ONLY (|vx|, |vy| <= eps in the MPC's feasible set). Rotates the 

9 body to the direction of travel -- or to the final pose heading 

10 once the position is reached. The MPC plans an accel-limited 

11 rotation that stops exactly on the target angle (terminal wz = 0), 

12 so there is no hand-tuned P/PD gain and no overshoot from late 

13 braking. Residual linear speed from a previous DRIVE is slewed to 

14 zero outside the MPC (the heading set forbids it). 

15 DRIVE : vx in [0, Vx] plus a SMALL |vy| <= vy_max (default 0.1) -- the vy 

16 budget only corrects small lateral drift while walking a straight 

17 line; wz is off (|wz| <= eps). The MPC drives straight at the 

18 target and decelerates to stop on it (terminal velocity 0). 

19 Because vy is tiny and wz is off, the optimiser cannot "cheat" 

20 with blended vy+wz arcs no matter how far the target is -- it just 

21 saturates vx, which IS the minimum-time policy for a straight line. 

22 

23A supervisor with hysteresis picks the state from the heading error to the 

24target bearing. Near the target the bearing is numerically unstable, so inside 

25`near_gate` only large errors can force HEADING; small drift is left to vy. 

26 

27Usage (per control tick): 

28 ctl = TurnDriveController(Vx=1.4, W=2.0, dt=0.1) 

29 out = ctl.step(x_robot, target_xy, u_meas=u_now, final_theta=goal_theta) 

30 apply(out["u"]) # send out["u"] to the low-level 

31 if out["done"]: ... # reached the target pose 

32"""

40""" 

41Build a light stability set that safely ALLOWS a small vy/wz blended with vx 

42during DRIVE. Idea: keep single-axis maxes (Vx,Vy,W) but couple vx so its budget 

43shrinks as vy/wz grow. Each coupling is ONE tunable linear cut per sign-quadrant: 

44 

45 a*vx + b*vy <= 1, a = 1/Vx, b = (1 - vx_cruise/Vx) / vy_allow 

46 

47so that: at vy=0 -> vx<=Vx (full); at vx=vx_cruise -> vy<=vy_allow; and vx is 

48derated in between (safe near the corner). Same for vx-wz with wz_allow. 

49 

50Tune vy_allow / wz_allow = how much lateral / yaw you want available while 

51cruising at vx_cruise. Keep them within the measured boundary. 

52(NOTE: kept for reference/experiments; the two-state controller below builds 

53its own per-state box constraints and does not use this.) 

54"""

57def make_blend_constraint( 

58 Vx: float, 

59 Vy: float, 

60 W: float, 

61 vx_cruise: float, 

62 vy_allow: float, 

63 wz_allow: float, 

64) -> np.ndarray: 

65 box = [ 

66 [1 / Vx, 0, 0], 

67 [-1 / Vx, 0, 0], 

68 [0, 1 / Vy, 0], 

69 [0, -1 / Vy, 0], 

70 [0, 0, 1 / W], 

71 [0, 0, -1 / W], 

72 ] 

73 a = 1 / Vx 

74 b = (1 - vx_cruise / Vx) / vy_allow # vx-vy coupling slope 

75 c = (1 - vx_cruise / Vx) / wz_allow # vx-wz coupling slope 

76 cuts = [] 

77 for sx in (1, -1): 

78 for sy in (1, -1): 

79 cuts.append([sx * a, sy * b, 0]) 

80 for sz in (1, -1): 

81 cuts.append([sx * a, 0, sz * c]) 

82 A = np.array(box + cuts) 

83 return A

86def _wrap(a: float) -> float: 

87 return np.arctan2(np.sin(a), np.cos(a))

90class TurnDriveController: 

91 def __init__( 

92 self, 

93 Vx: float, 

94 W: float, 

95 dt: float, 

96 N: int = 40, 

97 vy_max: float = 0.1, 

98 eps: float = 0.02, 

99 a_max: tuple[float, float, float] = (0.57, 0.3, 3.5), 

100 heading_enter_deg: float = 12.0, 

101 heading_exit_deg: float = 4.0, 

102 heading_exit_wz: float = 0.3, 

103 contact: float = 0.30, 

104 angle_tol: float = 0.1, 

105 near_gate: "float | None" = None, 

106 lag_tau_lin: float = 0.25, 

107 lag_tau_yaw: float = 0.25, 

108 w_run: float = 1.0, 

109 w_run_theta: float = 1.3, 

110 max_sqp_iter: int = 1, 

111 osqp_eps: float = 1e-3, 

112 osqp_max_iter: int = 1000, 

113 w_term: tuple[float, float, float] = (200, 200, 60), 

114 R: tuple[float, float, float] = (0.02, 0.12, 0.02), 

115 Rd: tuple[float, float, float] = (0.2, 0.3, 0.15), 

116 ) -> None: 

117 """ 

118 Vx : max forward speed [m/s] (DRIVE) 

119 W : max yaw rate [rad/s] (HEADING) 

120 vy_max : small lateral budget for drift correction during DRIVE [m/s] 

121 eps : "channel off" cap [m/s, rad/s]; vx/vy in HEADING, wz in DRIVE 

122 a_max : body accel box (ax, ay, az); also used to slew channels the 

123 active MPC does not own (e.g. braking vx when HEADING starts) 

124 heading_enter/exit_deg : supervisor hysteresis on the heading error. 

125 exit should be within what vy_max can correct while cruising 

126 (atan(vy_max/Vx) ~ 4 deg for 0.1/1.4). 

127 heading_exit_wz : also require |wz| below this to leave HEADING, so 

128 DRIVE (which clamps wz to eps) never cuts a rotation short. 

129 contact / angle_tol : done thresholds for position [m] / heading [rad] 

130 near_gate : within this distance the bearing is unstable; only errors 

131 > 45 deg may force HEADING (small drift is left to vy). 

132 lag_tau_lin / lag_tau_yaw : walk-controller actuation lag [s] for the 

133 linear (DRIVE) / yaw (HEADING) channels. The MPC plans from 

134 the state forward-predicted by this much with the measured 

135 rates, so braking starts one lag early instead of one lag 

136 late (the overshoot). Measure the true lag from logs as the 

137 phase delay between the command and the odometry velocity; 

138 0 disables. NOTE: set lag_tau_yaw to ~2x the measured lag -- 

139 the plant lags the braking RAMP too, which adds another 

140 wz*tau of carry-through on top of the instantaneous wz*tau 

141 the prediction covers (verified in closed-loop sim: with a 

142 0.4 s plant, 0.4 still overshoots ~34 deg, 0.8 -> 0 deg). 

143 DRIVE is position-feedback + stops `contact` early, so 

144 lag_tau_lin ~ the measured lag itself is enough. 

145 w_run / w_run_theta : progress costs = minimum-time front-loading for 

146 DRIVE position / HEADING rotation. Inside the braking 

147 envelope (d_gate / th_gate) they fade down to 

148 run_floor * w_run so the terminal approach is mostly 

149 quadratic-damped: fast far away, settles on the target pose 

150 with zero velocity and no overshoot (set run_floor = 0 for 

151 maximum robustness at the cost of a slow final approach). 

152 """ 

153 self.dt = float(dt) 

154 self.Vx, self.W, self.vy_max = float(Vx), float(W), float(vy_max) 

155 self.eps = float(eps) 

156 self.a_max = np.asarray(a_max, float) 

157 self.enter = np.radians(heading_enter_deg) 

158 self.exit = np.radians(heading_exit_deg) 

159 self.exit_wz = float(heading_exit_wz) 

160 self.contact = float(contact) 

161 self.angle_tol = float(angle_tol) 

162 self.near_gate = 2.0 * self.contact if near_gate is None else near_gate 

163 self.lag_tau_lin = float(lag_tau_lin) 

164 self.lag_tau_yaw = float(lag_tau_yaw) 

165 self.w_run = float(w_run) 

166 self.w_run_theta = float(w_run_theta) 

167 # progress-cost fade gates: outside the gate the progress cost makes 

168 # the profile bang-bang (minimum time), but bang-bang braking turns any 

169 # actuation lag directly into overshoot. Inside the braking envelope 

170 # (plus margin) the progress cost fades to zero, leaving the quadratic 

171 # terminal cost -- a damped, LQR-like approach (speed proportional to 

172 # the remaining error) that settles on the target with zero velocity 

173 # and no overshoot even with lag. Gate = 1.3 x the ideal bang-bang 

174 # braking distance from full speed. 

175 self.d_gate = 1.3 * self.Vx**2 / (2.0 * self.a_max[0]) 

176 self.th_gate = 1.3 * self.W**2 / (2.0 * self.a_max[2]) 

177 # inside the gate the progress cost keeps this fraction as a floor. 

178 # 0 = purely quadratic tail: zero overshoot even with NO lag 

179 # compensation, but the tail crawls (an N-step horizon with no arrival 

180 # incentive spreads the remaining distance over the whole horizon). 

181 # 0.2 keeps the tail at nearly full min-time speed and leaves the stop 

182 # to the lag prediction (closed-loop sim, 0.3 s plant: 3 cm overshoot 

183 # with matched lag_tau, 21 cm with lag_tau at HALF the true lag -- 

184 # still inside the 30 cm contact radius; over-estimating lag_tau 

185 # costs nothing, so err on the large side when tuning). 

186 self.run_floor = 0.2 

187 # per-state feasible sets (A @ u <= 1) 

188 self.A_head = np.array( 

189 [ 

190 [1 / eps, 0, 0], 

191 [-1 / eps, 0, 0], 

192 [0, 1 / eps, 0], 

193 [0, -1 / eps, 0], 

194 [0, 0, 1 / W], 

195 [0, 0, -1 / W], 

196 ], 

197 float, 

198 ) 

199 self.A_drive = np.array( 

200 [ 

201 [1 / Vx, 0, 0], 

202 [-1 / eps, 0, 0], # vx >= -eps 

203 [0, 1 / vy_max, 0], 

204 [0, -1 / vy_max, 0], 

205 [0, 0, 1 / eps], 

206 [0, 0, -1 / eps], 

207 ], 

208 float, 

209 ) 

210 common = { 

211 "N": N, 

212 "dt": dt, 

213 "a_max": tuple(a_max), 

214 "max_sqp_iter": max_sqp_iter, 

215 "osqp_eps": osqp_eps, 

216 "osqp_max_iter": osqp_max_iter, 

217 "w_term": w_term, 

218 "R": R, 

219 "Rd": Rd, 

220 } 

221 # Vx/Vy/W passed per planner are only warm-start / reachability hints; 

222 # the feasible set is poly_A. 

223 self._head = DiamondSQPPlanner( 

224 Vx=eps, Vy=eps, W=W, poly_A=self.A_head, **common 

225 ) 

226 self._drive = DiamondSQPPlanner( 

227 Vx=Vx, Vy=vy_max, W=eps, poly_A=self.A_drive, **common 

228 ) 

229 self.reset() 

230 

231 def reset(self) -> None: 

232 self.phase = "HEADING" 

233 self._zh = None # HEADING warm start 

234 self._zd = None # DRIVE warm start 

235 self._u = np.zeros(3) 

236 # arrived/done latches (hysteresis against boundary chatter, see step) 

237 self._arrived = False 

238 self._done = False

240 def _slew(self, u_prev: np.ndarray, u_des: np.ndarray) -> np.ndarray: 

241 d = self.a_max * self.dt 

242 return u_prev + np.clip(u_des - u_prev, -d, d)

244 @staticmethod 

245 def _project(u: np.ndarray, A: np.ndarray) -> np.ndarray: 

246 """Scale u onto the boundary of {A @ u <= 1} if it is outside, so the 

247 first-control accel box around u_current always intersects the set 

248 (otherwise the QP is infeasible and OSQP returns a divergent z).""" 

249 m = float(np.max(A @ u)) 

250 return u / m if m > 1.0 else u

252 def step( 

253 self, 

254 x: np.ndarray, 

255 target_xy: np.ndarray, 

256 u_meas: "np.ndarray | None" = None, 

257 wz_meas: "float | None" = None, 

258 v_meas: "np.ndarray | None" = None, 

259 final_theta: "float | None" = None, 

260 ) -> dict: 

261 """ 

262 x : (3,) robot pose (x, y, theta) [m, rad] 

263 target_xy : (2,) target position [m] 

264 u_meas : (3,) current body velocity command (for accel limiting); 

265 defaults to the last command issued 

266 wz_meas : measured yaw rate [rad/s] (e.g. odometry); when given it 

267 replaces u_meas[2] as the MPC's initial-rate constraint 

268 and drives the theta lag prediction 

269 v_meas : (2,) measured WORLD-frame linear velocity [m/s] (e.g. 

270 odometry); drives the position lag prediction. Falls 

271 back to the last command rotated into the world frame. 

272 final_theta : desired heading at the target pose; once the position is 

273 within `contact`, HEADING rotates to it (MPC in place) 

274 and done fires when |error| < angle_tol. If None, done 

275 fires on position alone. 

276 returns dict(u, phase, done, dist, heading_err, solve_ms) 

277 """ 

278 t0 = time.perf_counter() 

279 x = np.asarray(x, float) 

280 tgt = np.asarray(target_xy, float) 

281 u_prev = self._u if u_meas is None else np.asarray(u_meas, float) 

282 if wz_meas is not None: 

283 u_prev = np.array([u_prev[0], u_prev[1], float(wz_meas)]) 

284 

285 # actuation-lag compensation: the walk controller tracks commands with 

286 # a lag, so a plan made from the (lagging) measured pose brakes one 

287 # lag too late -> overshoot in both theta (HEADING) and x (DRIVE). 

288 # Plan instead from the state the robot will be in when this command 

289 # takes effect: forward-predict by tau with the measured rates 

290 # (fallback: the last command mapped to the world frame). done and the 

291 # supervisor also use the predicted state, so the stop is commanded 

292 # one lag early and the carry-through lands on the target. 

293 c, s = np.cos(x[2]), np.sin(x[2]) 

294 if v_meas is None: 

295 v_world = np.array( 

296 [u_prev[0] * c - u_prev[1] * s, u_prev[0] * s + u_prev[1] * c] 

297 ) 

298 else: 

299 v_world = np.asarray(v_meas, float) 

300 # position prediction, clipped so it never crosses the target: an 

301 # over-estimated lag_tau would otherwise predict PAST the target, flip 

302 # the bearing by 180 deg and trigger a spurious turnaround. 

303 step_w = v_world * self.lag_tau_lin 

304 d_raw = tgt - x[:2] 

305 dist_raw = float(np.linalg.norm(d_raw)) 

306 if dist_raw > 1e-9: 

307 dhat = d_raw / dist_raw 

308 along = float(step_w @ dhat) 

309 if along > dist_raw: 

310 step_w = step_w + (dist_raw - along) * dhat 

311 x = np.array([x[0] + step_w[0], x[1] + step_w[1], x[2]]) 

312 

313 d = tgt - x[:2] 

314 dist = float(np.linalg.norm(d)) 

315 bearing = np.arctan2(d[1], d[0]) 

316 # `arrived` is LATCHED with hysteresis (enter at contact, exit only at 

317 # 1.5x contact). DRIVE stops essentially ON the contact boundary, and 

318 # a biped rotating in place drifts a few cm: without the latch 

319 # `arrived` chatters every tick and the rotation target alternates 

320 # between final_theta and the bearing of a point 30 cm away (an 

321 # arbitrary direction) -- the observed final-alignment oscillation. 

322 if dist < self.contact: 

323 self._arrived = True 

324 elif dist > 1.5 * self.contact: 

325 self._arrived = False 

326 arrived = self._arrived 

327 want = final_theta if (arrived and final_theta is not None) else bearing 

328 # theta prediction, clipped so it never crosses the target angle: with 

329 # lag_tau_yaw larger than the true plant lag the raw prediction flips 

330 # the error sign every tick and HEADING limit-cycles around the 

331 # bearing (observed on an ideal plant with the 2x-lag setting: +/-7 

332 # deg oscillation, never converging). Clipping keeps the predicted 

333 # error the same sign as the raw error, so over-estimating the lag 

334 # degrades to "brake a bit early", never to oscillation. 

335 e_raw = _wrap(want - x[2]) 

336 dth = u_prev[2] * self.lag_tau_yaw 

337 dth = float(np.clip(dth, min(0.0, e_raw), max(0.0, e_raw))) 

338 x[2] = _wrap(x[2] + dth) 

339 e = _wrap(want - x[2]) 

340 

341 # done is LATCHED with a wider exit tolerance (1.5x angle_tol): after 

342 # the stop the braking carry-through can push |e| slightly back over 

343 # angle_tol, and a walking robot cannot realise arbitrarily small wz 

344 # corrections (footstep-quantised yaw) -- re-arming the rotation at 

345 # exactly angle_tol hunts forever around the tolerance edge. The latch 

346 # only releases if the error grows well past the tolerance (or the 

347 # position leaves the arrived hysteresis band), i.e. the target moved 

348 # or the robot was pushed. 

349 aligned = final_theta is None or abs(e) < self.angle_tol 

350 # require the rotation to be mostly braked before latching, otherwise 

351 # the carry-through immediately drifts the rest error to the edge of 

352 # the release band (the MPC is already braking to stop ON the angle, 

353 # so waiting costs at most a couple of ticks). 

354 if arrived and aligned and abs(u_prev[2]) < self.exit_wz: 

355 self._done = True 

356 elif not arrived or (final_theta is not None and abs(e) > 1.5 * self.angle_tol): 

357 self._done = False 

358 if self._done: 

359 u = self._slew(u_prev, np.zeros(3)) 

360 self._u = u 

361 return { 

362 "u": u, 

363 "phase": "DONE", 

364 "done": True, 

365 "dist": dist, 

366 "heading_err": e, 

367 "solve_ms": (time.perf_counter() - t0) * 1e3, 

368 } 

369 

370 # supervisor with hysteresis. Near the target the bearing is unstable: 

371 # inside near_gate only errors > 45 deg force HEADING (vy handles the 

372 # rest). Leaving HEADING additionally requires the rotation to be 

373 # almost braked (|wz| small), because DRIVE clamps wz to eps and would 

374 # otherwise cut the rotation short with an accel spike. 

375 if arrived: 

376 self.phase = "HEADING" # position reached: face final_theta 

377 elif ( 

378 self.phase == "DRIVE" 

379 and abs(e) > self.enter 

380 and (dist > self.near_gate or abs(e) > np.pi / 4) 

381 ): 

382 self.phase = "HEADING" 

383 elif ( 

384 self.phase == "HEADING" 

385 and abs(e) < self.exit 

386 and abs(u_prev[2]) < self.exit_wz 

387 ): 

388 self.phase = "DRIVE" 

389 

390 if self.phase == "HEADING": 

391 # rotation-only MPC: goal keeps the current position, terminal 

392 # wz = 0 stops exactly on the angle, w_run_theta front-loads the 

393 # turn (minimum time). Linear speed is not in this state's set: 

394 # slew it to zero outside the MPC. 

395 th_goal = x[2] + e # unwrapped absolute target heading 

396 u_cur = self._project(np.array([0.0, 0.0, u_prev[2]]), self.A_head) 

397 # fade the min-time cost inside the braking envelope -> damped 

398 # no-overshoot settling on the angle (see d_gate/th_gate in ctor) 

399 w_rt = self.w_run_theta * max( 

400 self.run_floor, 

401 min(1.0, max(0.0, (abs(e) - self.th_gate) / self.th_gate)), 

402 ) 

403 r = self._head.plan( 

404 x, 

405 [], 

406 np.array([x[0], x[1], th_goal]), 

407 goal_vel=np.zeros(3), 

408 u_current=u_cur, 

409 z_init=self._zh, 

410 w_run_theta=w_rt, 

411 ) 

412 self._zh = r["z"] 

413 self._zd = None # DRIVE warm start is stale now 

414 lin = self._slew(u_prev, np.zeros(3)) 

415 u = np.array([lin[0], lin[1], float(r["controls"][0][2])]) 

416 else: 

417 # straight-line MPC: forward vx + small vy, terminal velocity 0 

418 # (decelerate onto the target), w_run front-loads the progress 

419 # (minimum time). Goal theta = current theta: wz is off, so the 

420 # heading term must not fight the plan. 

421 u_cur = self._project( 

422 np.array( 

423 [ 

424 np.clip(u_prev[0], 0.0, self.Vx), 

425 np.clip(u_prev[1], -self.vy_max, self.vy_max), 

426 np.clip(u_prev[2], -self.eps, self.eps), 

427 ] 

428 ), 

429 self.A_drive, 

430 ) 

431 # fade the min-time cost inside the braking envelope -> damped 

432 # no-overshoot stop exactly on the target (see d_gate in ctor) 

433 w_r = self.w_run * max( 

434 self.run_floor, 

435 min(1.0, max(0.0, (dist - self.d_gate) / self.d_gate)), 

436 ) 

437 r = self._drive.plan( 

438 x, 

439 [], 

440 np.array([tgt[0], tgt[1], x[2]]), 

441 goal_vel=np.zeros(3), 

442 u_current=u_cur, 

443 z_init=self._zd, 

444 w_run=w_r, 

445 ) 

446 self._zd = r["z"] 

447 self._zh = None # HEADING warm start is stale now 

448 u = np.asarray(r["controls"][0], float) 

449 

450 self._u = u 

451 return { 

452 "u": u, 

453 "phase": self.phase, 

454 "done": False, 

455 "dist": dist, 

456 "heading_err": e, 

457 "solve_ms": (time.perf_counter() - t0) * 1e3, 

458 }
