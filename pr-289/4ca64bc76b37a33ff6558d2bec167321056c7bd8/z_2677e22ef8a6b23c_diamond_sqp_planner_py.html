1""" 

2SQP trajectory optimizer for an omnidirectional (vx, vy, wz) mobile/legged 

3robot: pass through given waypoints, reach a goal pose (x, y, theta), while 

4respecting the L1 velocity-command stability diamond 

5 |vx|/Vx + |vy|/Vy + |wz|/W <= 1. 

6 

7Formulation 

8 vars : states X = [x, y, theta]_{0..N}, controls U = [vx, vy, wz]_{0..N-1} 

9 model : x_{k+1} = x_k + dt * f(x_k, u_k) (body-frame kinematics) 

10 f = [vx cos th - vy sin th, vx sin th + vy cos th, wz] 

11 cost : quadratic -> Hessian P is CONSTANT (built once) 

12 control effort (anisotropic; vy penalised more) 

13 control smoothness (slew) 

14 waypoint position (soft) + terminal pose (soft) 

15 cons : kinematics (nonlinear equalities -> the only thing re-linearised) 

16 diamond per step (8 linear ineqs, constant) 

17 optional control slew-rate limit (linear) 

18 optional obstacle keep-out per knot (linearised separating 

19 hyperplane; radius inflated over time, leading knots only) 

20 

21SQP: each iteration solves ONE convex QP for the step dz with OSQP. 

22 - P constant; only the dynamics Jacobian (A) and the rhs (q,l,u) change. 

23 - symbolic factorisation done once; warm-started across iters and re-solves. 

24 - L1 merit line-search for globalisation; diamond stays feasible for any 

25 step in [0,1] because it is convex and both endpoints are feasible. 

26 

27Backend is swappable (default OSQP). Designed for weak CPUs (Jetson): 

28keep N modest, leave polish off, reuse the factorisation. 

29"""

118 def set_diamond(self, Vx: float, Vy: float, W: float) -> None: 

119 a = np.array([1 / Vx, 1 / Vy, 1 / W]) 

120 signs = np.array( 

121 [[sx, sy, sz] for sx in (1, -1) for sy in (1, -1) for sz in (1, -1)], float 

122 ) 

123 self.A_con = signs * a # (8,3): A_con @ u <= 1 

124 self.Vx, self.Vy, self.W = Vx, Vy, W

126 def set_polytope( 

127 self, 

128 A: np.ndarray, 

129 Vx: float | None = None, 

130 Vy: float | None = None, 

131 W: float | None = None, 

132 ) -> None: 

133 """General convex stability set as linear facets: A @ u <= 1 (K rows). 

134 Vx,Vy,W are the axis intercepts, used only for the warm-start guess / 

135 reachability estimate; if omitted they are read off from A.""" 

136 self.A_con = np.asarray(A, float) 

137 if Vx is None: # vx-axis intercept from facets 

138 Vx = 1.0 / np.max(self.A_con[:, 0].clip(min=1e-9)) 

139 Vy = 1.0 / np.max(self.A_con[:, 1].clip(min=1e-9)) 

140 W = 1.0 / np.max(self.A_con[:, 2].clip(min=1e-9)) 

141 self.Vx, self.Vy, self.W = Vx, Vy, W

143 # ---- kinematics ------------------------------------------------------- 

144 def _f(self, x: np.ndarray, u: np.ndarray) -> np.ndarray: 

145 th = x[2] 

146 c, s = np.cos(th), np.sin(th) 

147 return np.array([u[0] * c - u[1] * s, u[0] * s + u[1] * c, u[2]]) 

148 

149 def _AB(self, x: np.ndarray, u: np.ndarray) -> tuple[np.ndarray, np.ndarray]: 

150 th = x[2] 

151 c, s = np.cos(th), np.sin(th) 

152 A = np.zeros((3, 3)) 

153 A[0, 2] = -u[0] * s - u[1] * c 

154 A[1, 2] = u[0] * c - u[1] * s 

155 B = np.array([[c, -s, 0.0], [s, c, 0.0], [0.0, 0.0, 1.0]]) 

156 return A, B

159 def _build_cost(self) -> None: 

160 nz = self.nz 

161 Pr, Pc, Pv = [], [], [] 

162 qlin = np.zeros(nz) 

163 self._cost_targets = [] # filled per-plan (waypoints, goal)

165 def add( 

166 idxs: Sequence[int], 

167 coeffs: Sequence[float], 

168 weight: float, 

169 target: float = 0.0, 

170 ) -> None: 

171 for i, ci in zip(idxs, coeffs, strict=False): 

172 for j, cj in zip(idxs, coeffs, strict=False): 

173 Pr.append(i) 

174 Pc.append(j) 

175 Pv.append(weight * ci * cj) 

176 qlin[i] += -weight * target * ci 

177 

178 # control effort + slew (targets are 0 -> independent of plan) 

179 for k in range(self.N): 

180 ub = self.ui(k) 

181 for c in range(3): 

182 add([ub + c], [1.0], self.R[c]) 

183 if k >= 1: 

184 ub0 = self.ui(k - 1) 

185 for c in range(3): 

186 add([ub + c, ub0 + c], [1.0, -1.0], self.Rd[c]) 

187 # ridge 

188 for i in range(nz): 

189 Pr.append(i) 

190 Pc.append(i) 

191 Pv.append(self.reg) 

192 

193 self._P_static = sp.csr_matrix((Pv, (Pr, Pc)), shape=(nz, nz)) 

194 self._qlin_static = qlin

215 def add( 

216 idxs: Sequence[int], 

217 coeffs: Sequence[float], 

218 weight: float, 

219 target: float, 

220 ) -> None: 

221 for i, ci in zip(idxs, coeffs, strict=False): 

222 for j, cj in zip(idxs, coeffs, strict=False): 

223 Pr.append(i) 

224 Pc.append(j) 

225 Pv.append(weight * ci * cj) 

226 qlin[i] += -weight * target * ci 

227 

228 if self.soft_tracking: 

229 for w, k in zip(waypoints, wp_knots, strict=False): 

230 xb = self.xi(k) 

231 add([xb + 0], [1.0], self.w_wp, w[0]) 

232 add([xb + 1], [1.0], self.w_wp, w[1]) 

233 xbN = self.xi(self.N) 

234 for c in range(3): 

235 add([xbN + c], [1.0], self.w_term[c], goal[c]) 

236 # running progress cost: pull every intermediate knot toward the goal 

237 # so the plan closes the gap early (favours time-to-goal). w_run pulls 

238 # position (x, y); w_run_theta pulls heading so rotation overlaps the 

239 # approach (no slow in-place spin at the end). 

240 if w_run > 0.0 or w_run_theta > 0.0: 

241 for k in range(1, self.N): 

242 xb = self.xi(k) 

243 add([xb + 0], [1.0], w_run, goal[0]) 

244 add([xb + 1], [1.0], w_run, goal[1]) 

245 add([xb + 2], [1.0], w_run_theta, goal[2]) 

246 

247 # terminal velocity (kick): drive the last controls to goal_vel 

248 if goal_vel is not None: 

249 for k in range(self.N - vel_hold, self.N): 

250 ub = self.ui(k) 

251 for c in range(3): 

252 add([ub + c], [1.0], self.w_goal_vel, goal_vel[c]) 

253 

254 P = self._P_static + sp.csr_matrix((Pv, (Pr, Pc)), shape=(nz, nz)) 

255 return P.tocsr(), qlin

257 # ---- obstacles --------------------------------------------------------- 

258 def _norm_obstacles( 

259 self, 

260 obstacles: Sequence[Sequence[float]] | None, 

261 ) -> list[tuple[np.ndarray, float, float, np.ndarray | None]]: 

262 """Normalise obstacle specs to [(center(2,), radius, inflate, ns|None)]. 

263 Each spec is (x, y, radius), (x, y, radius, inflate) or 

264 (x, y, radius, inflate, nsx, nsy). 

265 inflate: keep-out growth rate [m/s] for THIS obstacle (worst-case 

266 opponent speed). Strategy intent lives here too: 0 for a 

267 contested opponent we accept contact with, larger to give a 

268 fast/dangerous opponent a wide berth. Missing -> ctor 

269 obs_inflate. 

270 (nsx, nsy): world-frame normal forcing which side to pass on: 

271 ns @ (p - o) >= 0.""" 

272 out: list[tuple[np.ndarray, float, float, np.ndarray | None]] = [] 

273 for ob in obstacles or []: 

274 ob = np.asarray(ob, float).ravel() 

275 infl = float(ob[3]) if ob.size >= 4 else self.obs_inflate 

276 ns = None 

277 if ob.size >= 6: 

278 n = np.linalg.norm(ob[4:6]) 

279 if n > 1e-9: 

280 ns = ob[4:6] / n 

281 out.append((ob[:2].copy(), float(ob[2]), infl, ns)) 

282 return out

285 """Duration [s] used for the first-step acceleration bound. Falls back to 

286 the plan step dt when the caller gives nothing or a non-positive / 

287 non-finite value (so a garbage measurement can never open up the bound).""" 

288 if u_current_dt is None: 

289 return self.dt 

290 dt0 = float(u_current_dt) 

291 if not np.isfinite(dt0) or dt0 <= 0.0: 

292 return self.dt 

293 return dt0

295 # ---- constraint assembly for the step dz (A, l, u) ------------------- 

296 def _constraints( 

297 self, 

298 z: np.ndarray, 

299 x_start: np.ndarray, 

300 waypoints: Sequence[Sequence[float]] | None = None, 

301 wp_knots: Sequence[int] | None = None, 

302 goal: np.ndarray | None = None, 

303 u_current: np.ndarray | None = None, 

304 obstacles: list[tuple[np.ndarray, float, np.ndarray | None]] | None = None, 

305 u_current_dt: float | None = None, 

306 ) -> tuple[sp.csc_matrix, np.ndarray, np.ndarray, float]: 

307 N, dt = self.N, self.dt 

308 X = z[: self.nx * (N + 1)].reshape(N + 1, 3) 

309 U = z[self.nx * (N + 1) :].reshape(N, 3) 

310 rows, cols, data, lo, up = [], [], [], [], [] 

311 r = 0 

312 

313 def ent(rr: int, cc: int, vv: float) -> None: 

314 rows.append(rr) 

315 cols.append(cc) 

316 data.append(vv) 

317 

318 # initial condition: dx_0 = x_start - x_0 

319 for i in range(3): 

320 ent(r + i, self.xi(0) + i, 1.0) 

321 lo += list(x_start - X[0]) 

322 up += list(x_start - X[0]) 

323 r += 3 

324 

325 # dynamics defects, linearised 

326 g_inf = 0.0 

327 for k in range(N): 

328 A, B = self._AB(X[k], U[k]) 

329 g = X[k + 1] - X[k] - dt * self._f(X[k], U[k]) 

330 g_inf = max(g_inf, np.max(np.abs(g))) 

331 for i in range(3): 

332 ent(r + i, self.xi(k) + i, -1.0) # -I (dx_k) 

333 ent(r + i, self.xi(k + 1) + i, 1.0) # +I (dx_{k+1}) 

334 ent(r + 0, self.xi(k) + 2, -dt * A[0, 2]) # -dt A (theta col) 

335 ent(r + 1, self.xi(k) + 2, -dt * A[1, 2]) 

336 for i in range(3): # -dt B (du_k) 

337 for j in range(3): 

338 if B[i, j] != 0.0 or (i, j) in [ 

339 (0, 0), 

340 (0, 1), 

341 (1, 0), 

342 (1, 1), 

343 (2, 2), 

344 ]: 

345 if (i, j) in [(0, 0), (0, 1), (1, 0), (1, 1), (2, 2)]: 

346 ent(r + i, self.ui(k) + j, -dt * B[i, j]) 

347 lo += list(-g) 

348 up += list(-g) 

349 r += 3 

350 

351 # stability set per control: A_con (U_k + du_k) <= 1 (K facets) 

352 K = self.A_con.shape[0] 

353 for k in range(N): 

354 ub = self.A_con @ U[k] 

355 for row in range(K): 

356 for j in range(3): 

357 ent(r + row, self.ui(k) + j, self.A_con[row, j]) 

358 lo.append(-np.inf) 

359 up.append(1.0 - ub[row]) 

360 r += K 

361 

362 # optional slew-rate limit: -du_max <= (U_k+du_k)-(U_{k-1}+du_{k-1}) <= du_max 

363 if self.du_max is not None: 

364 for k in range(1, N): 

365 diff = U[k] - U[k - 1] 

366 for c in range(3): 

367 ent(r, self.ui(k) + c, 1.0) 

368 ent(r, self.ui(k - 1) + c, -1.0) 

369 lo.append(-self.du_max[c] - diff[c]) 

370 up.append(self.du_max[c] - diff[c]) 

371 r += 1 

372 

373 # body acceleration limit: |(U_k+du_k)-(U_{k-1}+du_{k-1})| <= a_max*dt. 

374 # If u_current (the robot's current command / measured velocity) is given, 

375 # also bound the very first command: 

376 # |(U_0+du_0)-u_current| <= a_max*dt_first (moving-target re-plan). 

377 # dt_first is u_current_dt when given, else the plan step dt. The horizon 

378 # steps use dt (that IS their duration), but the first step bridges the 

379 # real world: u_current was applied for one CONTROL PERIOD, which is only 

380 # dt when the caller really runs at 1/dt. A caller running slower must pass 

381 # its measured period, otherwise the achievable acceleration scales with the 

382 # loop rate (at half rate the robot accelerates at half a_max) and the 

383 # command crawls up the ramp in wall-clock time. 

384 if self.a_max is not None: 

385 da = self.a_max * dt 

386 if u_current is not None: 

387 da0 = self.a_max * self._first_step_dt(u_current_dt) 

388 for c in range(3): 

389 ent(r, self.ui(0) + c, 1.0) 

390 diff0 = U[0][c] - u_current[c] 

391 lo.append(-da0[c] - diff0) 

392 up.append(da0[c] - diff0) 

393 r += 1 

394 for k in range(1, N): 

395 diff = U[k] - U[k - 1] 

396 for c in range(3): 

397 ent(r, self.ui(k) + c, 1.0) 

398 ent(r, self.ui(k - 1) + c, -1.0) 

399 lo.append(-da[c] - diff[c]) 

400 up.append(da[c] - diff[c]) 

401 r += 1 

402 

403 # obstacle keep-out, linearised at z: for knots k = 1..obs_kmax require 

404 # n_ik @ (p_k + dp_k - o_i) >= r_i + obs_inflate * k*dt 

405 # with n_ik the unit vector from the obstacle to the current-iterate 

406 # knot position (separating hyperplane; re-linearised every SQP iter). 

407 # The required outward move is capped by what the robot can reach in 

408 # k steps, so a violated start (e.g. an obstacle popping up on top of 

409 # us) degrades to "push out as fast as possible" instead of making the 

410 # QP infeasible. Row count is fixed per (obstacle, knot) so the OSQP 

411 # sparsity signature stays stable while the obstacle count is stable. 

412 # a violated row may only demand what the plant can actually recover in 

413 # k steps (~ a_lat t^2 / 4, half the ideal bang-bang displacement); 

414 # asking for more makes the QP infeasible and OSQP's max-iter iterate 

415 # is garbage that would blow up the warm-start chain. 

416 a_rec = min(self.a_max[0], self.a_max[1]) if self.a_max is not None else None 

417 for o, rad, infl, ns in obstacles or []: 

418 # passing axis: obstacle -> goal. Stable w.r.t. the robot's own 

419 # position (goal - x_start degenerates the moment we arrive and 

420 # left stale side rows active behind us -> infeasible QP). 

421 # Contest case (obstacle ~ at the goal): fall back to 

422 # start -> obstacle, then to the heading. 

423 gd = (o * 0.0) if goal is None else (goal[:2] - o) 

424 if np.linalg.norm(gd) < 0.3: 

425 gd = o - x_start[:2] 

426 ngd = np.linalg.norm(gd) 

427 if ngd < 1e-6: 

428 gd = np.array([np.cos(x_start[2]), np.sin(x_start[2])]) 

429 ngd = 1.0 

430 gd = gd / ngd 

431 gperp = np.array([-gd[1], gd[0]]) # left of the passing axis 

432 # passing-side direction used to break the dead-ahead symmetry: 

433 # exactly on the axis the radial normal is longitudinal, the QP has 

434 # no first-order incentive to sidestep, and the robot parks in 

435 # front of the disc. Tilting the normal toward the passing side 

436 # (only inside the disc's lateral shadow) restores that gradient. 

437 # Side = the strategy's ns when given, else the side the current 

438 # iterate already leans to (warm-start chain => stays latched), 

439 # left when perfectly ambiguous (deterministic, cannot dither). 

440 if ns is not None: 

441 bias_dir = ns 

442 else: 

443 lat_sum = sum( 

444 float(gd[0] * (X[k, 1] - o[1]) - gd[1] * (X[k, 0] - o[0])) 

445 for k in range(1, self.obs_kmax + 1) 

446 ) 

447 bias_dir = gperp if lat_sum >= 0.0 else -gperp 

448 for k in range(1, self.obs_kmax + 1): 

449 dvec = X[k, :2] - o 

450 d = np.linalg.norm(dvec) 

451 r_k = rad + infl * k * dt 

452 lat = abs(float(gd[0] * dvec[1] - gd[1] * dvec[0])) 

453 shadow = max(0.0, 1.0 - lat / max(r_k, 0.05)) 

454 n = dvec / d if d > 1e-9 else bias_dir 

455 if ns is not None and float(n @ ns) < 0.0: 

456 # radial push toward the forbidden side would fight the 

457 # side rows (slow zigzag); mirror it onto the allowed side. 

458 n = n - 2.0 * float(n @ ns) * ns 

459 n = n + 0.6 * shadow * bias_dir 

460 n = n / np.linalg.norm(n) 

461 t_k = k * dt 

462 cap = ( 

463 0.25 * a_rec * t_k * t_k 

464 if a_rec is not None 

465 else 0.5 * t_k * self.Vx 

466 ) 

467 ent(r, self.xi(k) + 0, n[0]) 

468 ent(r, self.xi(k) + 1, n[1]) 

469 lo.append(min(r_k - float(n @ dvec), cap)) 

470 up.append(np.inf) 

471 r += 1 

472 if ns is not None and float(gd @ dvec) < r_k + 0.3: 

473 # strategy-forced side: ns @ (p_k - o) >= 0. Only while the 

474 # knot has not clearly passed the obstacle along the 

475 # approach axis -- afterwards the side no longer means 

476 # anything and the row only makes the QP harder. 

477 g_s = float(ns @ dvec) 

478 ent(r, self.xi(k) + 0, ns[0]) 

479 ent(r, self.xi(k) + 1, ns[1]) 

480 lo.append(min(-g_s, cap)) 

481 up.append(np.inf) 

482 r += 1 

483 

484 # hard tracking equalities (waypoint positions + goal pose) 

485 if not self.soft_tracking and waypoints is not None: 

486 for w, k in zip(waypoints, wp_knots, strict=False): 

487 for c in range(2): 

488 ent(r, self.xi(k) + c, 1.0) 

489 val = w[c] - X[k, c] 

490 lo.append(val) 

491 up.append(val) 

492 r += 1 

493 for c in range(3): 

494 ent(r, self.xi(N) + c, 1.0) 

495 val = goal[c] - X[N, c] 

496 lo.append(val) 

497 up.append(val) 

498 r += 1 

499 

500 A = sp.csc_matrix((data, (rows, cols)), shape=(r, self.nz)) 

501 return A, np.array(lo), np.array(up), g_inf 

502 

503 # ---- merit / cost helpers -------------------------------------------- 

504 def _cost_rel(self, z: np.ndarray, P: sp.spmatrix, q: np.ndarray) -> float: 

505 return 0.5 * z @ (P @ z) + q @ z 

506 

507 def _dyn_viol(self, z: np.ndarray, x_start: np.ndarray) -> float: 

508 N, dt = self.N, self.dt 

509 X = z[: self.nx * (N + 1)].reshape(N + 1, 3) 

510 U = z[self.nx * (N + 1) :].reshape(N, 3) 

511 v = np.sum(np.abs(X[0] - x_start)) 

512 for k in range(N): 

513 v += np.sum(np.abs(X[k + 1] - X[k] - dt * self._f(X[k], U[k]))) 

514 return v 

515 

516 # ---- initial guess ---------------------------------------------------- 

517 def _init_guess( 

518 self, 

519 x_start: np.ndarray, 

520 waypoints: Sequence[Sequence[float]], 

521 wp_knots: Sequence[int], 

522 goal: np.ndarray, 

523 ) -> tuple[np.ndarray, float]: 

524 N = self.N 

525 pts_k = [0, *list(wp_knots), N] 

526 pts_p = [x_start[:2]] + [np.asarray(w) for w in waypoints] + [goal[:2]] 

527 X = np.zeros((N + 1, 3)) 

528 # piecewise-linear position through knots 

529 for seg in range(len(pts_k) - 1): 

530 k0, k1 = pts_k[seg], pts_k[seg + 1] 

531 for k in range(k0, k1 + 1): 

532 a = 0 if k1 == k0 else (k - k0) / (k1 - k0) 

533 X[k, :2] = (1 - a) * pts_p[seg] + a * pts_p[seg + 1] 

534 # theta: linear from start to (unwrapped) goal 

535 th_goal = x_start[2] + np.arctan2( 

536 np.sin(goal[2] - x_start[2]), np.cos(goal[2] - x_start[2]) 

537 ) 

538 X[:, 2] = np.linspace(x_start[2], th_goal, N + 1) 

539 # controls by inverse kinematics of the position guess (body frame) 

540 U = np.zeros((N, 3)) 

541 for k in range(N): 

542 dp = (X[k + 1, :2] - X[k, :2]) / self.dt 

543 th = X[k, 2] 

544 c, s = np.cos(th), np.sin(th) 

545 U[k, 0] = c * dp[0] + s * dp[1] 

546 U[k, 1] = -s * dp[0] + c * dp[1] 

547 U[k, 2] = (X[k + 1, 2] - X[k, 2]) / self.dt 

548 # project into the stability set if needed 

549 r = self.A_con @ U[k] 

550 m = np.max(r) 

551 if m > 1.0: 

552 U[k] /= m * 1.001 

553 return np.concatenate([X.ravel(), U.ravel()]), th_goal 

554 

555 # ---- main entry ------------------------------------------------------- 

556 def plan( 

557 self, 

558 x_start: np.ndarray, 

559 waypoints: Sequence[Sequence[float]], 

560 goal: np.ndarray, 

561 goal_vel: np.ndarray | None = None, 

562 wp_knots: Sequence[int] | None = None, 

563 z_init: np.ndarray | None = None, 

564 u_current: np.ndarray | None = None, 

565 time_budget: float | None = None, 

566 w_run: float = 0.0, 

567 w_run_theta: float = 0.0, 

568 obstacles: Sequence[Sequence[float]] | None = None, 

569 u_current_dt: float | None = None, 

570 ) -> dict[str, Any]: 

571 """ 

572 x_start : (3,) current pose 

573 waypoints: list of (x, y) to pass through (soft) 

574 goal : (3,) target pose (x, y, theta) 

575 goal_vel: optional (3,) BODY-frame velocity (vx,vy,wz) at the final knot, 

576 e.g. (v_kick, 0, 0) to arrive moving forward along the heading 

577 (a kick): set goal[2] to the kick direction and this to (v,0,0). 

578 wp_knots: optional knot index per waypoint (default: by chord fraction) 

579 z_init : optional warm-start full vector (e.g. shifted previous plan) 

580 u_current: optional (3,) BODY-frame current velocity (the last command, or 

581 better the measured one); with a_max set, bounds the first planned 

582 command to |u0 - u_current| <= a_max*u_current_dt so a 

583 moving-target re-plan cannot jump the command (no accel spike). 

584 u_current_dt: wall-clock duration [s] the u_current bound is allowed to 

585 accelerate over, i.e. the caller's CONTROL PERIOD (not the plan 

586 step dt). Defaults to dt, which only matches reality when the 

587 caller really closes the loop at 1/dt: a caller at half that rate 

588 and 

589 leaving this None gets half the acceleration in wall-clock time 

590 (the ramp is per tick, not per second). Pass the measured loop 

591 period (clamped to something sane) to make the ramp 

592 rate-independent. Non-positive / non-finite -> dt. 

593 time_budget: optional wall-clock cap [s]. When the elapsed solve time 

594 reaches it, the SQP loop stops and returns the best-effort z 

595 found so far (warm-started next call). Bounds worst-case latency. 

596 w_run: optional running-cost weight pulling every intermediate knot's 

597 (x, y) toward the goal. > 0 front-loads the motion so the plan 

598 reaches the goal earlier (favours time-to-goal). Larger = more 

599 aggressive; lower it if tracking becomes unstable. 

600 w_run_theta: optional running-cost weight pulling every intermediate 

601 knot's heading toward the goal, so rotation overlaps the 

602 approach instead of a slow in-place spin at the end. 

603 obstacles: optional list of keep-out discs in the WORLD frame, each 

604 (x, y, radius), (x, y, radius, inflate) or 

605 (x, y, radius, inflate, nsx, nsy). radius is the 

606 strategy-chosen keep-out (intent: ~0 to contest, large to 

607 avoid wide); it grows by inflate [m/s] over the first 

608 obs_horizon seconds (worst-case motion, no velocity 

609 estimate needed; inflate defaults to ctor obs_inflate, use 

610 0 for an opponent we accept contact with). (nsx, nsy) 

611 optionally forces which side to pass on: ns @ (p - o) >= 0. 

612 returns dict(states, controls, feasible, iters, cost, g_inf, time, 

613 diamond_use, obs_margin) 

614 """ 

615 t0 = time.perf_counter() 

616 x_start = np.asarray(x_start, float) 

617 goal = np.asarray(goal, float) 

618 if goal_vel is not None: 

619 goal_vel = np.asarray(goal_vel, float) 

620 if u_current is not None: 

621 u_current = np.asarray(u_current, float) 

622 obs = self._norm_obstacles(obstacles) 

623 N = self.N 

624 

625 if wp_knots is None: 

626 pts = [x_start[:2]] + [np.asarray(w) for w in waypoints] + [goal[:2]] 

627 seg = np.linalg.norm(np.diff(pts, axis=0), axis=1) 

628 cum = np.concatenate([[0], np.cumsum(seg)]) 

629 frac = cum / cum[-1] 

630 wp_knots = [round(f * N) for f in frac[1:-1]] 

631 wp_knots = [min(max(k, 1), N - 1) for k in wp_knots] 

632 # reachability: each leg needs >= dist / (Vx * dt) steps 

633 need = int(np.ceil(cum[-1] / (self.Vx * self.dt))) 

634 if need > N: 

635 print_in_strategy( 

636 f" ! horizon may be too short: need ~{need} steps for " 

637 f"{cum[-1]:.2f} m at Vx={self.Vx}, have N={N}. " 

638 "Increase N or dt, or expect best-effort feasibility." 

639 ) 

640 

641 # unwrap goal theta once, build cost 

642 if z_init is not None: 

643 z = z_init.copy() 

644 th_goal = x_start[2] + np.arctan2( 

645 np.sin(goal[2] - x_start[2]), np.cos(goal[2] - x_start[2]) 

646 ) 

647 else: 

648 z, th_goal = self._init_guess(x_start, waypoints, wp_knots, goal) 

649 goal_unwrapped = np.array([goal[0], goal[1], th_goal]) 

650 P, q_lin = self._cost_with_targets( 

651 waypoints, 

652 wp_knots, 

653 goal_unwrapped, 

654 goal_vel=goal_vel, 

655 w_run=w_run, 

656 w_run_theta=w_run_theta, 

657 ) 

658 P_tri = sp.triu(P).tocsc() 

659 

660 iters = 0 

661 feas_count = 0 

662 prev_cost = None 

663 osqp_status = "not run" 

664 solver_ok = False 

665 for it in range(self.max_sqp_iter): 

666 # wall-clock deadline: stop and return the best-effort z so far. 

667 if time_budget is not None and time.perf_counter() - t0 >= time_budget: 

668 break 

669 iters = it + 1 

670 A, lo, up, g_inf = self._constraints( 

671 z, 

672 x_start, 

673 waypoints, 

674 wp_knots, 

675 goal_unwrapped, 

676 u_current, 

677 obs, 

678 u_current_dt, 

679 ) 

680 q = P @ z + q_lin # gradient at z (for the step) 

681 

682 res = None 

683 sig = (A.shape[0], A.data.size, P_tri.data.size) 

684 if self._osqp is not None and sig != getattr(self, "_sig", None): 

685 self._osqp = None # problem structure changed -> rebuild 

686 if self._osqp is None: 

687 self._osqp = osqp.OSQP() 

688 self._osqp.setup( 

689 P_tri, 

690 q, 

691 A, 

692 lo, 

693 up, 

694 warm_start=True, 

695 verbose=self.verbose, 

696 polish=False, 

697 eps_abs=self.osqp_eps, 

698 eps_rel=self.osqp_eps, 

699 max_iter=self.osqp_max_iter, 

700 adaptive_rho=True, 

701 ) 

702 self._sig = sig 

703 else: 

704 # same sparsity pattern -> in-place numeric update, no re-symbolic 

705 self._osqp.update(q=q, l=lo, u=up, Ax=A.data, Px=P_tri.data) 

706 

707 res = self._osqp.solve() 

708 dz = res.x 

709 osqp_status = res.info.status 

710 if dz is None or not np.all(np.isfinite(dz)): 

711 solver_ok = False 

712 break # genuine solver failure 

713 # Iterate acceptance. The real-robot polytope QPs routinely hit 

714 # "maximum iterations reached" even warm (measured: most ticks) -- 

715 # those iterates are nearly feasible (prim_res ~1e-3..3e-2) and 

716 # merely suboptimal, which RTI tolerates: rejecting them stalls the 

717 # warm-start chain and the robot crawls at a_max*dt. What must be 

718 # rejected is the DIVERGED iterate (infeasible problem, prim_res 

719 # orders of magnitude larger; applying it once corrupted the plan 

720 # to +100 m). Gate on the primal residual, not on optimality. 

721 solver_ok = osqp_status in ("solved", "solved inaccurate") or ( 

722 osqp_status == "maximum iterations reached" 

723 and float(res.info.prim_res) < 5e-2 

724 ) 

725 if not solver_ok: 

726 break 

727 step = np.max(np.abs(dz)) 

728 

729 # globalisation: near feasibility trust the linearisation (full step); 

730 # far from it, damp with an L1 merit line search. 

731 if g_inf < 10 * self.tol_feas: 

732 alpha = 1.0 

733 else: 

734 mu = 1e3 

735 phi0 = self._cost_rel(z, P, q_lin) + mu * self._dyn_viol(z, x_start) 

736 alpha = 1.0 

737 for _ in range(25): 

738 zt = z + alpha * dz 

739 phit = self._cost_rel(zt, P, q_lin) + mu * self._dyn_viol( 

740 zt, x_start 

741 ) 

742 if phit <= phi0: 

743 break 

744 alpha *= 0.5 

745 z = z + alpha * dz 

746 

747 # stopping: dynamics consistent + (tiny step OR cost settled), with 

748 # a short confirmation so we exit a couple of iters after feasibility. 

749 cost = self._cost_rel(z, P, q_lin) 

750 settled = prev_cost is not None and abs(cost - prev_cost) <= 1e-4 * ( 

751 1 + abs(prev_cost) 

752 ) 

753 prev_cost = cost 

754 if g_inf < self.tol_feas: 

755 feas_count += 1 

756 if step < self.tol_step or settled or feas_count >= 3: 

757 break 

758 else: 

759 feas_count = 0 

760 

761 # restoration: a failed solve on a warm start means the carried plan 

762 # has gone stale beyond repair (e.g. an obstacle popped up inside it 

763 # while we kept rejecting iterates -> the frozen plan stays infeasible 

764 # forever). Retry ONCE from a fresh straight-line guess at the current 

765 # state; if that also fails the caller must brake (solver_ok tells). 

766 if not solver_ok and z_init is not None: 

767 budget_left = ( 

768 None 

769 if time_budget is None 

770 else max(time_budget - (time.perf_counter() - t0), 1e-3) 

771 ) 

772 return self.plan( 

773 x_start, 

774 waypoints, 

775 goal, 

776 goal_vel=goal_vel, 

777 wp_knots=wp_knots, 

778 z_init=None, 

779 u_current=u_current, 

780 time_budget=budget_left, 

781 w_run=w_run, 

782 w_run_theta=w_run_theta, 

783 obstacles=obstacles, 

784 u_current_dt=u_current_dt, 

785 ) 

786 

787 X = z[: self.nx * (N + 1)].reshape(N + 1, 3) 

788 U = z[self.nx * (N + 1) :].reshape(N, 3) 

789 use = (U @ self.A_con.T).max(axis=1) # max facet value per step (<=1) 

790 _, _, _, g_inf = self._constraints( 

791 z, 

792 x_start, 

793 waypoints, 

794 wp_knots, 

795 goal_unwrapped, 

796 u_current, 

797 obs, 

798 u_current_dt, 

799 ) 

800 # worst signed clearance over the constrained knots (>= 0 means the 

801 # inflated keep-outs are respected; slightly < 0 can happen while the 

802 # capped recovery is still pushing out of a violated start). 

803 obs_margin = np.inf 

804 for o, rad, infl, _ns in obs: 

805 for k in range(1, self.obs_kmax + 1): 

806 d = float(np.linalg.norm(X[k, :2] - o)) 

807 obs_margin = min(obs_margin, d - (rad + infl * k * self.dt)) 

808 return { 

809 "states": X, 

810 "controls": U, 

811 "z": z, 

812 "feasible": bool(g_inf < 1e-3), 

813 "iters": iters, 

814 "cost": float(self._cost_rel(z, P, q_lin)), 

815 "g_inf": float(g_inf), 

816 "diamond_use": use, 

817 "obs_margin": float(obs_margin), 

818 "osqp_status": osqp_status, 

819 "solver_ok": bool(solver_ok), 

820 "time": time.perf_counter() - t0, 

821 "wp_knots": wp_knots, 

822 "goal_unwrapped": goal_unwrapped, 

823 }
