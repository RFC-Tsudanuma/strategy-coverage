183def _build_sqp_planner(far: bool) -> "DiamondSQPPlanner": 

184 global _poly_A 

185 from diamond_sqp_backend import DiamondSQPPlanner 

186 

187 d = np.load(_STABILITY_POLYTOPE_NPZ) 

188 _poly_A = d["A"] 

189 poly_A = d["A"] * 0.8 # A @ u <= 1 の安定多面体（b は全て 1） 

190 if far: 

191 # |vy| <= GOTO_FAR_VY_MAX を A @ u <= 1 の行として追加（他成分は不変）。 

192 vy_rows = np.array( 

193 [[0.0, 1.0 / GOTO_FAR_VY_MAX, 0.0], [0.0, -1.0 / GOTO_FAR_VY_MAX, 0.0]] 

194 ) 

195 poly_A = np.vstack([poly_A, vy_rows]) 

196 _sqp_planner_poly_A[far] = poly_A 

197 return DiamondSQPPlanner( 

198 # N=20（2秒先・dt=0.1）。MPC は先頭制御 u0 しか使わないので 2 秒で十分。 

199 # N=40 比で計算量を約 1/3.7 に削減（u0 はほぼ不変）し、実機 10Hz を確保する。 

200 N=_SQP_N, 

201 dt=_SQP_DT, 

202 Vx=float(d["Vx"]), 

203 Vy=float(d["Vy"]), 

204 W=float(d["W"]), 

205 poly_A=poly_A, 

206 osqp_max_iter=1000, # 単発QPの暴走を抑える（毎tickのwarm startで十分収束する） 

207 # RTI（Real-Time Iteration）：1tick=SQP1反復に固定。毎tickの warm start 連鎖で 

208 # 数tickかけて収束する実時間MPCの定石。1反復に固定することで計算量を更に削減し、 

209 # 実機での制御周期を確保する（収束しきらない初回tickのみu0がやや粗いが安定領域内）。 

210 max_sqp_iter=1, 

211 # 速度成分の効果重み (vx, vy, wz)。vy を高くし vy 混合を抑える。 

212 # wz=1.0: w_run（進捗コスト）は「vy の加速度予算を ax に上乗せし、 

213 # 和ベクトルがゴールを向くよう wz で体を回す」エクスプロイトを誘発し、 

214 # 直進ですら常時回転しながら歩く（実測 max|wz|=1.1 rad/s）。wz の 

215 # 効果コストを上げると巡航中の無意味な回転が消え（max|wz|=0.27）、 

216 # 到達も速くなる。最終姿勢への正当な大回転（π 回頭）は w_run_theta の 

217 # 勾配が勝るので影響しない（実測: 所要時間不変）。 

218 R=(0.02, 0.50, 0.05), 

219 # wz の加速度上限を下げ、1tick の wz 変化（=theta 制御入力の跳ね）を抑えて 

220 # 歩行を安定化する（vx,vy は既定のまま）。max|Δwz| = GOTO_AMAX_WZ*dt。 

221 a_max=list(GOTO_A_MAX), 

222 )

242 # 計画に必要な地平長（N=20 の 2 秒では制動だけで足りない）。

245 # 実機の加速度上限（DiamondSQPPlanner 既定と同値）。MPC の初回制御の 

246 # 拘束と、状態遷移時に MPC が持たないチャネルを落とす slew の両方に効く。

258 # 線形は位置フィードバック + contact 手前停止が効くので実測 τ 程度でよい。 

259 # まだ回頭が行き過ぎるなら lag_tau_yaw を、手前で止まるなら下げる方向。

282def _u_current_from_pub( 

283 pub: WalkCommandPublisher, dt: float, rotate: bool = True 

284) -> "np.ndarray | None": 

285 """直前に送った歩行コマンド（ボディ系 vx,vy[cm/s], wz[rad/s]）を SQP の 

286 u_current（ボディ系 m/s, rad/s）へ変換して返す。a_max が U0 を実指令に対して 

287 拘束し、controls[0] が直前コマンドから急変しないようにする。 

288 

289 回転補正(rotate=True): 直前コマンドの線形速度は「前tickの機体向き」のボディ系だが、今tick 

290 までに機体は約 wz*dt 回頭している。線形速度を -wz*dt 回転して今tickのボディ系へ合わせる。 

291 高速回頭中に横歩きを強制されるのを避ける目的でキック接近用に導入した。 

292 ただし go_to のような大回頭では、回頭中(wz大)に大きな横成分を注入して a_max 経由で wz を

294 ため向きの素直さが要る呼び出しでは rotate=False（回転補正なし＝raw）を使う（120°ちょうどで収束）。 

295 wz は角速度なのでフレーム不変。未送信なら None（従来通り horizon 内差分のみ拘束）。 

296 

297 多面体射影: 回転後の (vx, vy) は安定多面体 A@u<=1 を僅かに超えることがある。超えた点を 

298 u_current にすると a_max ボックスが多面体と交わらず QP が実行不能→OSQP 発散解で warm start 

299 連鎖を汚染する。これを防ぐため、超えた場合は原点方向へスケールして多面体境界へ射影する。""" 

300 cmd = pub.current_walk_cmd 

301 if cmd is None: 

302 return None 

303 vx = cmd.x_velocity / 100.0 

304 vy = cmd.y_velocity / 100.0 

305 wz = cmd.theta_velocity

309 # 前tickボディ系の (vx, vy) を -dth 回転 → 今tickボディ系へ。

313 if _poly_A is not None: 

314 m = float(np.max(_poly_A @ u)) 

315 if m > 1.0: 

316 u = u / m 

317 return u

389def _u_current_for_plan( 

390 s: StrategyState, pub: WalkCommandPublisher, far: bool 

391) -> "np.ndarray | None": 

392 """プランナ初段のアンカー u_current（ボディ系 m/s, rad/s）を返す。プランナはこれを 

393 |u0 - u_current| <= a_max * u_current_dt として拘束する（箱の幅は呼び出し側が渡す 

394 実ループ周期で決まり、ここでは決めない）。 

395 

396 アンカーは「直前に *計画した* 速度指令」（_sqp_last_planned_u）。pub の直前指令では 

397 なく計画側の値を持つのは、安全側の停止指令（発散検知・目標喪失・他アクション）が 

398 1tick 挟まっても指令列の連鎖を切らないため。指令をアンカーにしていると、混入した 

399 停止でランプがゼロから再スタートし（10Hz なら次の指令は最大 4cm/s）歩行が這う。 

400 計画履歴が無い初回だけ pub の直前指令、それも無ければ実測速度で代用する。 

401 

402 実測速度（odom）の役割は「アンカーが現実から離れすぎるのを防ぐ」ことだけ: 

403 アンカーを [u_meas ± a_max*(dt_loop + GOTO_ODOM_LAG_S)] へクリップする。一次遅れ 

404 のプラントでは指令が実速度を a_max*tau 先行するのが正常なので、この幅の中に居る 

405 限りクリップは効かない（＝通常走行では従来と完全に同じ挙動）。効くのは実機が指令に 

406 全く追従していない場合（壁や他機に阻まれている、転倒明け、odom 基準で明らかに 

407 別の速度で歩いている）で、指令が現実から乖離して暴走するのを止める。 

408 

409 アンカーをこの幅で「広げる」のではなく「クリップする」のが要点。幅を広げると 

410 a_max のスルーレート制限そのものが緩み、1tick の wz 跳ねが設計値 

411 （GOTO_AMAX_WZ*周期）を超えて歩行安定性を損なう（実測: |Δwz| が 0.12 rad/s 設計に 

412 対し 0.165 rad/s まで開いた）。""" 

413 u = _sqp_last_planned_u 

414 if u is None: 

415 u = _u_current_from_pub(pub, _SQP_DT, rotate=False) 

416 u_meas = _u_meas_from_odom(s) 

417 if u is None: 

418 u = u_meas 

419 if u is None: 

420 return None 

421 u = np.array(u, float) 

422 if u_meas is not None: 

423 lead = np.asarray(GOTO_A_MAX, float) * ( 

424 _SQP_DT + np.asarray(GOTO_ODOM_LAG_S, float) 

425 ) 

426 u = np.clip(u, u_meas - lead, u_meas + lead) 

427 return _project_anchor_into_stability_polytope(u, far)

530def sqp_go_to_target_global_pose( 

531 s: StrategyState, 

532 pub: WalkCommandPublisher, 

533 target: "state.Position", 

534 goal_vel: "np.ndarray | None" = None, 

535 heading_gate_dist_cm: "float | None" = GOTO_HEADING_GATE_DIST_CM, 

536) -> WalkCommand | None: 

537 """ 

538 Action to go to a target pose in debug mode. 

539 DiamondSQPPlanner で安定領域（速度指令ダイヤモンド）を制約に速度指令を計画し、 

540 その先頭の制御 u0 = (vx, vy, wz)[ボディ座標系] を歩行コマンドとして送る。 

541 

542 target を渡すとそのグローバル姿勢へ向かう。省略時はユーザ指定の 

543 pub.target_pose_from_user を使う（従来の HTN からの呼び出し互換）。 

544 heading_gate_dist_cm: 向き距離ゲート [cm]。遠いうちは進行方向を向いて前進し、近づくと 

545 target.theta へブレンドする（道中の横歩きを避ける）。同じゲートで遠いうちは横速度も 

546 |vy| <= GOTO_FAR_VY_MAX にハード制限し（far プランナ切替）、横移動での位置合わせは 

547 ゲート内に入ってからにする。None で無効（vy 制限も無し）。自前で向きを決める 

548 呼び出し（via_goto の Phase A 等）は None を渡して opt-out する。 

549 goal_vel（ボディ系 [vx,vy,wz]）を渡すと終端速度をそこへ拘束する。[0,0,0] を渡すと 

550 終端で停止する計画になり、目標到達時の速度（特に wz）が 0 に減速するので、姿勢合わせの 

551 オーバーシュート（行き過ぎ→±180°付近で最短方向が反転して無駄な一回転）を防げる。 

552 """ 

553 print_in_strategy("[Debug mode]: Going to target pose (SQP).") 

554 

555 cur = s.self_position_global 

556 cfg = DEFAULT_STRATEGY_CONFIG["common_config"] 

557 

558 # 目標に十分近く、姿勢も合っていれば停止（既存の move_to_target_global_pose と同じ閾値）。 

559 dist_cm = math.hypot(target.x - cur.x, target.y - cur.y) # 位置は cm 

560 dtheta = abs( 

561 math.atan2( 

562 math.sin(target.theta - cur.theta), math.cos(target.theta - cur.theta) 

563 ) 

564 ) 

565 if ( 

566 dist_cm < cfg["move_pose_no_move_distance"] 

567 and dtheta < cfg["move_pose_no_move_angle"] 

568 ):

570 # （float64 フィールド）へ int が入り、publish 時に rosidl の C 変換 

571 # (geometry_msgs__msg__vector3__convert_from_py の PyFloat_Check) が失敗して 

572 # ノードが abort する。theta_velocity は send_walk_command で単位変換されず 

573 # そのまま Twist へ入るため、int リテラルだと確実に落ちる。 

574 return WalkCommand(x_velocity=0.0, y_velocity=0.0, theta_velocity=0.0) 

575 

576 # SQP は m, m/s, rad で計画する。状態（self_position_global）は cm なので m に変換する。 

577 x_start = np.array([cur.x / 100.0, cur.y / 100.0, cur.theta]) 

578 # 向き距離ゲート: 遠いうちは進行方向（ゴール位置への方位）を向いて前進し、近づくと 

579 # target.theta（最終姿勢）へブレンドする。二足の道中の90度横歩き/後ろ歩きを避け、到達時に回頭。 

580 goal_theta = target.theta 

581 if heading_gate_dist_cm is not None: 

582 bearing = math.atan2(target.y - cur.y, target.x - cur.x) 

583 w = min(1.0, dist_cm / heading_gate_dist_cm) 

584 goal_theta = target.theta + w * math.atan2( 

585 math.sin(bearing - target.theta), math.cos(bearing - target.theta) 

586 ) 

587 goal = np.array([target.x / 100.0, target.y / 100.0, goal_theta]) 

588 

589 global _sqp_warm_z, _sqp_warm_target, _sqp_far_mode, _sqp_last_planned_u 

590 target_key = (target.x, target.y, target.theta) 

591 z_init = _sqp_warm_z if target_key == _sqp_warm_target else None 

592 

593 # 距離ゲートで far/near プランナを切替（ヒステリシス付き）。遠いうちは |vy| を 

594 # ハード制限した far プランナで「回頭して前進」に寄せ、ゲート内に入ったら通常 

595 # プランナで横移動も使った最終位置合わせを許す。向き距離ゲートを opt-out した 

596 # 呼び出し（heading_gate_dist_cm=None）は自前で向きを管理するので常に near。 

597 if heading_gate_dist_cm is None: 

598 _sqp_far_mode = False 

599 elif _sqp_far_mode: 

600 _sqp_far_mode = dist_cm > heading_gate_dist_cm 

601 else: 

602 _sqp_far_mode = dist_cm > heading_gate_dist_cm * GOTO_FAR_SWITCH_HYST 

603 planner = _get_sqp_planner(far=_sqp_far_mode) 

604 # 初段の加速度制限に使う実ループ周期 [s]。プランナ既定はプラン刻み _SQP_DT 固定なので、 

605 # 実ループがそれより遅い環境（計算資源の乏しい実機で他タスクと同居する等）では 

606 # 1 tick の許容速度変化が周期に依らず a_max*_SQP_DT のままになり、実効加速度が 

607 # 周期比で落ちる（10Hz→3.3Hz で vx 0.40→0.13 m/s^2。閉ループシムで 2 秒後の指令が 

608 # 81→29 cm/s、4 秒の到達距離 254→94 cm ＝目標へ向かう歩行が明らかに遅い）。実測 

609 # ループ周期（前回 tick の所要時間）を渡して壁時計時間ベースにする。下限は _SQP_DT 

610 # なので 10Hz 以上で回れている間の挙動は従来と同一。 

611 dt_loop = float( 

612 np.clip( 

613 s.last_strategy_process_time, 

614 _SQP_DT, 

615 _SQP_DT * GOTO_U_CURRENT_MAX_DT_SCALE, 

616 ) 

617 ) 

618 # 初段のアンカー（直前に計画した指令。odom で現実から離れすぎないようクリップ）。 

619 u_current = _u_current_for_plan(s, pub, far=_sqp_far_mode) 

620 # 制御周期 10Hz=100ms。最悪計算時間を 50ms で頭打ちにし、打ち切り時は 

621 # best-effort 解を返す（次tickは warm start から継続して収束する）。 

622 # w_run: 途中ノードを goal 位置へ引く進捗コスト。到達時間を重視して動きを 

623 # 前倒しする。歩きが不安定なら下げる（速度/加速度はハード制約で常に安全側）。 

624 # w_run_theta: 姿勢も歩きながら合わせ、最後のその場回転を無くす。0.05 では回頭の 

625 # 勾配が弱く、初期回頭が終端コスト任せで後ろ倒しになる（地平いっぱいに回転を 

626 # 引き延ばす）。0.15 で回頭が前倒しされ、大回頭（90〜180°）の初期 turn が速くなる 

627 # （閉ループシム: 方位 60〜180° 平均で t80 3.28→3.10 秒、直進時の無意味な回転は 

628 # 増えない）。上げ過ぎると横歩き経由の到達が遅くなるケースがあるので少しずつ。 

629 result = planner.plan( 

630 x_start, 

631 [], 

632 goal, 

633 goal_vel=goal_vel, 

634 z_init=z_init, 

635 u_current=u_current, 

636 u_current_dt=dt_loop, 

637 time_budget=0.05, 

638 w_run=1.0, 

639 w_run_theta=0.15, 

640 ) 

641 if not _solution_sane(result): 

642 # 発散解: warm start を汚染しないよう破棄（次tickは cold）、安全のため停止。 

643 print_in_strategy("[Debug mode]: SQP solution diverged; stopping (go_to).") 

644 _sqp_warm_z = None 

645 return WalkCommand(x_velocity=0.0, y_velocity=0.0, theta_velocity=0.0) 

646 _sqp_warm_z = result["z"] 

647 _sqp_warm_target = target_key 

648 

649 # 計画した経路を可視化用にパブリッシュする。 

650 # SQP の states はグローバル m（先頭は現在位置）。visualizer はローカル cm として 

651 # 解釈し先頭にロボット位置を付けるので、先頭を除いてローカル cm に変換して送る。 

652 waypoints_local = [ 

653 convert_global_to_local( 

654 cur, 

655 state.Position( 

656 state.Position.COORD_GLOBAL, px * 100.0, py * 100.0, float(ptheta) 

657 ), 

658 ) 

659 for px, py, ptheta in result["states"][1:] 

660 ] 

661 pub.send_planned_path(waypoints_local) 

662 

663 # 先頭の制御（ボディ座標系の速度指令）。プランナの解は未収束 tick では初段の箱や 

664 # 多面体を超え得るので、送信前にハードにクリップする（_clamp_command_to_first_step_box）。 

665 # m/s -> cm/s に戻して送る。wz は rad/s のまま。 

666 u_send = _clamp_command_to_first_step_box( 

667 result["controls"][0], u_current, dt_loop, far=_sqp_far_mode 

668 ) 

669 # 次tickの初段アンカー。安全側の停止指令（発散検知・目標喪失・他アクション）で 

670 # 指令列が切れてもランプを維持するため、実際に送った指令ではなく計画値を持つ 

671 # （詳細は _u_current_for_plan）。 

672 _sqp_last_planned_u = u_send 

673 # numpy スカラのまま WalkCommand に入れない（float 派生なので publish は通るが、 

674 # json 化や比較で型が漏れる）。Python の float にして返す。 

675 vx, vy, wz = (float(c) for c in u_send) 

676 pub.send_target_position(target) # デバッグ表示のために目標姿勢を送る 

677 return WalkCommand(vx * 100.0, vy * 100.0, wz) # 歩行コマンドを返す

692 debug_go_to_target_pose の TurnDriveController 版（方式比較用）。 

693 2状態 MPC（HEADING: wz のみで進行方向へ回頭 / DRIVE: 前進 vx + 誤差修正用の 

694 小さな vy のみで直進）を毎tick解いて時間最短で目標姿勢へ向かう。位置到達後は 

695 HEADING が target.theta へその場回頭し、位置・角度の両閾値を満たしたら停止する。 

696 

697 go_to との相違: goal_vel 引数は無い（DRIVE が常に終端速度 0 で減速停止）。また 

698 TurnDriveController は計画列を公開しないため経路可視化（send_planned_path）は 

699 行わない。 

700 """ 

701 print_in_strategy("[Debug mode]: Going to target pose (turn-drive).") 

702 logger.action_executed(details="Goto target pose in DEBUG mode (turn-drive)") 

703 

704 if target is None: 

705 target = pub.target_pose_from_user 

706 if target is None: 

707 pub.send_stop_command() 

708 return s 

709 

710 cur = s.self_position_global 

711 cfg = DEFAULT_STRATEGY_CONFIG["common_config"] 

712 ctl = _get_turn_drive_controller() 

713 

714 # 目標が変わったら状態（HEADING/DRIVE）と warm start をリセットする（go_to が 

715 # 目標変更で warm start を捨てるのと同方針）。 

716 global _turn_drive_target 

717 target_key = (target.x, target.y, target.theta) 

718 if target_key != _turn_drive_target: 

719 ctl.reset() 

720 _turn_drive_target = target_key 

721 

722 dist_cm = math.hypot(target.x - cur.x, target.y - cur.y) 

723 e_theta = math.atan2( 

724 math.sin(target.theta - cur.theta), math.cos(target.theta - cur.theta) 

725 ) 

726 if ( 

727 dist_cm < cfg["move_pose_no_move_distance"] 

728 and abs(e_theta) < cfg["move_pose_no_move_angle"] 

729 ): 

730 pub.send_stop_command() 

731 return s 

732 

733 # 直前コマンド（ボディ系 m/s, 回転補正済み）。状態ごとの制約セットへの射影は 

734 # TurnDriveController が内部で行うのでここでは射影しない。 

735 u_meas = _u_current_from_pub(pub, _SQP_DT) 

736 # 実測速度（odom 由来）。wz [rad/s] は HEADING の初期レート拘束と θ の遅れ 

737 # 前進予測に、線形速度（ワールド系 cm/s -> m/s）は位置の遅れ前進予測に使う。 

738 # 未取得なら None → 直前の指令値で代用。 

739 if s.robot_velocity is None: 

740 wz_meas = None 

741 v_meas = None 

742 else: 

743 wz_meas = float(s.robot_velocity.theta_vel) 

744 v_meas = [ 

745 s.robot_velocity.x_vel / 100.0, 

746 s.robot_velocity.y_vel / 100.0, 

747 ] 

748 

749 # SQP は m, rad で計画する。状態（cm）を m に変換して 1 tick 進める。 

750 # final_theta により位置到達後は HEADING が target.theta へその場回頭する。 

751 x_m = np.array([cur.x / 100.0, cur.y / 100.0, cur.theta]) 

752 out = ctl.step( 

753 x_m, 

754 [target.x / 100.0, target.y / 100.0], 

755 u_meas=u_meas, 

756 wz_meas=wz_meas, 

757 v_meas=v_meas, 

758 final_theta=target.theta, 

759 )

761 # 発散解の遮断（_solution_sane と同じ基準を先頭制御に適用）。 

762 if not np.all(np.isfinite(u)) or ( 

763 float(np.max(_turn_drive_poly_A @ u)) > _SOLUTION_POLY_SANE_MAX 

764 ): 

765 print_in_strategy( 

766 "[Debug mode]: turn-drive solution diverged; stopping (go_to)." 

767 ) 

768 ctl.reset() 

769 pub.send_stop_command() 

770 return s 

771 

772 # ボディ座標系の速度指令。m/s -> cm/s に戻して送る。wz は rad/s のまま。 

773 pub.send_target_position(target) 

774 pub.send_walk_command(WalkCommand(u[0] * 100.0, u[1] * 100.0, u[2])) 

775 return s

778@contract.htn_action(

783def debug_kick_ball_to_goal_via_goto( 

784 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

785) -> None: 

786 """ 

787 性能比較用の別方式キック。SQP キック（debug_kick_ball_to_goal）と違い、専用の 

788 キック計画は持たず debug_go_to_target_pose（通常の go_to）だけで実現する。 

789 

790 Phase A（接近）：ボール->ゴール直線上の、ボールからゴールと反対側 100cm の姿勢 

791 （＝キック方向を向く）へ go_to で向かう。 

792 Phase B（キック）：その姿勢に到達したら、単にボール位置を目標にして go_to で 

793 ボールへ直進する。歩いてボールを押し出すことで蹴る。 

794 ボールが大きく動いたら再接近する。 

795 """ 

796 print_in_strategy("[Debug mode]: Kicking ball to goal via go_to (compare).") 

797 logger.action_executed(details="Kick ball to goal via go_to in DEBUG mode") 

798 

799 if s.world_states["ball_position_relative"] == BALL_POSITION.UNKNOWN: 

800 pub.send_stop_command() 

801 return s 

802 

803 cur = s.self_position_global 

804 ball_global = convert_local_to_global(cur, s.ball_position_local) 

805 goal_pos_global = convert_local_to_global(cur, get_goal_true_center_pos(s)) 

806 # ボール->ゴール中心のグローバル方位（キック方向）。 

807 kick_direction_global = math.atan2( 

808 goal_pos_global.y - ball_global.y, goal_pos_global.x - ball_global.x 

809 ) 

810 kick_dir = (math.cos(kick_direction_global), math.sin(kick_direction_global)) 

811 

812 # アプローチ姿勢（cm）：ボールからゴールと反対方向へ 100cm、キック方向を向く。 

813 approach_x = ball_global.x - kick_dir[0] * KICK_GOTO_APPROACH_BACK_CM 

814 approach_y = ball_global.y - kick_dir[1] * KICK_GOTO_APPROACH_BACK_CM 

815 

816 # フェーズ判定：一度アプローチ姿勢に到達したらキックへコミット（突き抜け中も維持）。 

817 # ボールが大きく動いたら（新しい対象）解除して再接近する。 

818 global _goto_kick_committed, _goto_kick_commit_key 

819 ball_now = (ball_global.x, ball_global.y) 

820 if _goto_kick_commit_key is not None and ( 

821 math.hypot( 

822 ball_now[0] - _goto_kick_commit_key[0], 

823 ball_now[1] - _goto_kick_commit_key[1], 

824 ) 

825 > KICK_RETARGET_DIST_CM 

826 ): 

827 _goto_kick_committed = False 

828 _goto_kick_commit_key = ball_now 

829 

830 dist_to_approach_cm = math.hypot(approach_x - cur.x, approach_y - cur.y) 

831 dtheta = abs( 

832 math.atan2( 

833 math.sin(kick_direction_global - cur.theta), 

834 math.cos(kick_direction_global - cur.theta), 

835 ) 

836 ) 

837 # アプローチ姿勢に到達したらキックへコミット。go_to は到達で停止するので、位置・角度が 

838 # 揃った時点で移行する（速度ゲートは付けない：instantaneous 速度が odom ノイズ等で 15cm/s 

839 # 未満に落ちないと恒久的にコミットできず停止したままになるため。Phase B は単純なラムなので 

840 # 多少勢いが残っていてもボールへ向かって突っ込むだけで問題ない）。 

841 if ( 

842 dist_to_approach_cm < KICK_APPROACH_REACH_M * 100.0 

843 and dtheta < KICK_APPROACH_ANGLE_RAD 

844 ): 

845 _goto_kick_committed = True 

846 

847 if _goto_kick_committed: 

848 # Phase B（キック）：SQP も go_to も使わず、ボールへ真っ直ぐ歩いて突っ込む。 

849 # ボディ系のボール方位へ wz で向き続けながら前進速度でラムする。 

850 to_ball = math.atan2(ball_global.y - cur.y, ball_global.x - cur.x) 

851 bearing_body = math.atan2( 

852 math.sin(to_ball - cur.theta), math.cos(to_ball - cur.theta) 

853 ) 

854 wz = max( 

855 -KICK_RAM_YAW_MAX, min(KICK_RAM_YAW_MAX, KICK_RAM_YAW_KP * bearing_body) 

856 ) 

857 pub.send_walk_command(WalkCommand(KICK_RAM_SPEED_CMS, 0.0, wz)) 

858 # 可視化：ボール位置にキック方向の矢印。 

859 pub.send_target_position( 

860 state.Position( 

861 state.Position.COORD_GLOBAL, 

862 ball_global.x, 

863 ball_global.y, 

864 kick_direction_global, 

865 ) 

866 ) 

867 return s 

868 

869 # Phase A：ボール裏 100cm のアプローチ姿勢へ go_to。接近中の目標向きは 

870 # KICK_GOTO_APPROACH_HEADING_MODE で切替（どちらも approach 到達時は kick 方向へ収束）。 

871 # この向き制御は via_goto の Phase A でのみ効く（go_to 本体や他アクションには影響しない）。 

872 if KICK_GOTO_APPROACH_HEADING_MODE == "face_travel": 

873 # 進行方向（approach 地点への方位）を向く＝横歩き/無駄回転を最小化。近づくにつれ 

874 # kick 方向へブレンド（到達時は kick 方向）。 

875 bearing_travel = math.atan2(approach_y - cur.y, approach_x - cur.x) 

876 w_travel = min(1.0, dist_to_approach_cm / KICK_GOTO_HEADING_BLEND_DIST_CM) 

877 d_head = math.atan2( 

878 math.sin(bearing_travel - kick_direction_global), 

879 math.cos(bearing_travel - kick_direction_global), 

880 ) 

881 goal_theta = kick_direction_global + w_travel * d_head 

882 else: # "face_ball" 

883 # ボールを向く（キック方向からボール方位側へ KICK_GOTO_FACE_BALL_BLEND 寄せる）。 

884 # approach 姿勢ではボール方位==キック方向なので終端は kick 方向。 

885 bearing_to_ball = math.atan2(ball_global.y - cur.y, ball_global.x - cur.x) 

886 d_head = math.atan2( 

887 math.sin(bearing_to_ball - kick_direction_global), 

888 math.cos(bearing_to_ball - kick_direction_global), 

889 ) 

890 goal_theta = kick_direction_global + KICK_GOTO_FACE_BALL_BLEND * d_head 

891 target = state.Position( 

892 state.Position.COORD_GLOBAL, approach_x, approach_y, goal_theta 

893 ) 

894 # 終端速度 0 で到達させ、姿勢合わせのオーバーシュート（→無駄な一回転）を防ぐ。 

895 # 向き距離ゲートは opt-out（ここで自前に goal_theta を決めているため）。 

896 return debug_go_to_target_pose( 

897 s, pub, logger, target=target, goal_vel=np.zeros(3), heading_gate_dist_cm=None 

898 )
