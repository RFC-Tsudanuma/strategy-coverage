59# 観測してから LAST_VISIBLE_BALL_STALE_SEC 以内なら NEAR_FIRST モードを選ぶ。 

60NEAR_FIRST_RADIUS_CM = 150.0 

61LAST_VISIBLE_BALL_STALE_SEC = 10.0

64@contract.htn_action() 

65def do_flontier_exploration( 

66 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

67) -> StrategyState | None: 

68 """ 

69 フロンティア探索を行うアクション。 

70 last_visible_ball が新しく自分から NEAR_FIRST_RADIUS_CM 以内なら NEAR_FIRST モード、 

71 それ以外（遠い／観測なし／古すぎ）は FAR_FIRST モードでフロンティア探索を開始する。 

72 フロンティア探索を停止するのは、アクションを使った実装だと煩雑になるので、使わずに停止させても良い。 

73 """ 

74 print_in_real_action("Executing do_flontier_exploration().") 

75 if not action_functions.need_flontier_exploration(s): 

76 logger.action_precondition_not_satisfied( 

77 "No need to perform flontier_exploration" 

78 ) 

79 return None 

80 

81 last_ball, last_ts = s.last_visible_ball 

82 if ( 

83 last_ball is not None 

84 and (get_strategy_time() - last_ts) <= LAST_VISIBLE_BALL_STALE_SEC 

85 and calculate_distance_global(s.self_position_global, last_ball) 

86 <= NEAR_FIRST_RADIUS_CM 

87 ): 

88 mode = DoFlontierExploration.Request.MODE_EXPLORE_NEAR_FIRST 

89 mode_label = "NEAR_FIRST" 

90 else: 

91 mode = DoFlontierExploration.Request.MODE_EXPLORE_FAR_FIRST 

92 mode_label = "FAR_FIRST" 

93 

94 # 複数ロボットでの役割分担: 奇数 ID は時計回り、偶数 ID は反時計回り。 

95 # FILL_FEET 時は flontier_exp 側で NONE 固定にしているのでここでは方向だけを渡す。 

96 if s.robot_id % 2 == 1: 

97 rotation_direction = DoFlontierExploration.Request.ROTATION_CW 

98 rotation_label = "CW" 

99 else: 

100 rotation_direction = DoFlontierExploration.Request.ROTATION_CCW 

101 rotation_label = "CCW" 

102 

103 pub.do_flontier_exploration( 

104 robot_id=s.robot_id, mode=mode, rotation_direction=rotation_direction 

105 ) 

106 logger.action_executed( 

107 details=f"Performing frontier exploration (mode={mode_label}, rot={rotation_label})" 

108 ) 

109 return s

165# コーナーの分担から外すロール。 

166# KEEPER はゴール前に留まるので、そもそもコーナーを見に行かない。 

167# NEUTRAL はペナルティ中・停止中など「動いてはいけない/動けない」状態なので、

171CORNERKICK_EXCLUDED_ROLES = frozenset({ROLE.KEEPER, ROLE.NEUTRAL})

174def get_cornerkick_watch_robot_ids(s: StrategyState) -> list[int]: 

175 """ 

176 コーナーキック時にコーナーの確認を分担する、フィールド上にいるロボットIDを昇順で返す。 

177 CORNERKICK_EXCLUDED_ROLES のロールの味方は分担からは外す。 

178 情報共有が CORNERKICK_SHARED_INFO_STALE_SEC 以内に届いている味方のみを 

179 「フィールドに存在する」とみなす。 

180 

181 自分は常に含める。このアクションに到達する時点で 

182 need_flontier_exploration が真であり、その条件に role != NEUTRAL が 

183 含まれているので、自分が除外対象のロールになることはない。 

184 """ 

185 # 情報共有がこの秒数以内に届いている味方を「フィールドに存在する」とみなす 

186 CORNERKICK_SHARED_INFO_STALE_SEC = 2.0 

187 watch_ids = {s.robot_id} 

188 for ally in s.shared_msg: 

189 if ally is None or ally.role in CORNERKICK_EXCLUDED_ROLES: 

190 continue 

191 if not utils.shared_info.shared_info_is_within_second( 

192 ally, CORNERKICK_SHARED_INFO_STALE_SEC 

193 ): 

194 continue 

195 watch_ids.add(ally.robot_id) 

196 return sorted(watch_ids)

199def get_cornerkick_watch_pose( 

200 s: StrategyState, corner_y_sign: float, offset_step: int 

201) -> Position: 

202 """ 

203 コーナーを確認するために立つ位置(グローバル座標)を計算する。 

204 

205 立ち位置の基準はゴールエリアの角。そこからIDの順番に応じて 

206 コーナーから離れる向き(ゴールエリアの角がコーナーから見て内側にある向き)へずらす。 

207 

208 :param corner_y_sign: 担当するコーナーのy符号(+1なら左、-1なら右) 

209 :param offset_step: 同じコーナーを担当するロボットの中でのID順。 

210 この段数だけ基準からコーナーより離れた位置に立つ 

211 コーナーキックのボールは守備側のゴールライン上に置かれるので、 

212 自チームのコーナーキックなら敵陣ゴールライン、相手のコーナーキックなら自陣ゴールライン側を見る。 

213 座標は「自陣が左側」の前提で計算し、最後に実際のour_fieldへ変換する。 

214 """ 

215 

216 # cornerkick_exploration で使う定数。 

217 # コーナーを確認するために立つ位置はゴールエリアの角が基準で、 

218 # 同じコーナーを担当するロボットが複数いる場合はIDの順番に応じて 

219 # CORNERKICK_WATCH_OFFSET_STEP_CM ずつコーナーから離れる向きへずらす。 

220 CORNERKICK_WATCH_OFFSET_STEP_CM = 150.0 

221 # 立ち位置がフィールド外へはみ出さないようにするための、フィールド端からのマージン 

222 CORNERKICK_WATCH_FIELD_MARGIN_CM = 50.0 

223 is_our_corner_kick = s.kickOffTeam == setting.get_team_id() 

224 goal_line_sign = 1.0 if is_our_corner_kick else -1.0 

225 corner_x = goal_line_sign * setting.FIELD_DIMENTIONS.X_LENGTH / 2.0 

226 corner_y = corner_y_sign * setting.FIELD_DIMENTIONS.Y_LENGTH / 2.0 

227 

228 # 立ち位置の基準はコーナーと同じ側のゴールエリアの角 

229 base_x = goal_line_sign * setting.FIELD_DIMENTIONS.GOAL_AREA_X_FRONT 

230 base_y = corner_y_sign * setting.FIELD_DIMENTIONS.GOAL_AREA_WIDTH / 2.0 

231 

232 # コーナーからゴールエリアの角へ向かう向き(=コーナーから離れる向き)に、 

233 # IDの順番に応じたオフセット分だけ基準からずらす 

234 away_x = base_x - corner_x 

235 away_y = base_y - corner_y 

236 away_norm = math.hypot(away_x, away_y) 

237 if away_norm > 0.0: 

238 away_x /= away_norm 

239 away_y /= away_norm 

240 offset = CORNERKICK_WATCH_OFFSET_STEP_CM * offset_step 

241 x_limit = setting.FIELD_DIMENTIONS.X_LENGTH / 2.0 - CORNERKICK_WATCH_FIELD_MARGIN_CM 

242 y_limit = setting.FIELD_DIMENTIONS.Y_LENGTH / 2.0 - CORNERKICK_WATCH_FIELD_MARGIN_CM 

243 stand_x = clamp(base_x + away_x * offset, -x_limit, x_limit) 

244 stand_y = clamp(base_y + away_y * offset, -y_limit, y_limit) 

245 

246 # コーナー(ボールが置かれている場所)を向く 

247 theta = math.atan2(corner_y - stand_y, corner_x - stand_x) 

248 return convert_hardcoded_pos_to_our_field( 

249 s, 

250 Position(coord=Position.COORD_GLOBAL, x=stand_x, y=stand_y, theta=theta), 

251 )

254def _cornerkick_search_theta_velocity(s: StrategyState) -> float: 

255 """ 

256 立ち位置に着いた後、その場探索で回り続けるときの旋回速度(rad/s)を返す。 

257 

258 速度は generate_turn_command_by_angle の微調整領域と同じ 

259 ROBOT_MAX_SPEED_PI * turn_min_vel_scale を使う(自己位置がぶれない程度に遅い)。 

260 向きは do_flontier_exploration と同じ規約で、奇数 ID は時計回り、 

261 偶数 ID は反時計回りにして、同じコーナーを見る機体同士の死角を減らす。 

262 """ 

263 speed = float( 

264 setting.ROBOT_MAX_SPEED_PI 

265 * setting.DEFAULT_STRATEGY_CONFIG["common_config"]["turn_min_vel_scale"] 

266 ) 

267 return -speed if s.robot_id % 2 == 1 else speed

277def cornerkick_exploration( 

278 s: StrategyState, pub: WalkCommandPublisher, logger: DecisionLogger 

279) -> StrategyState | None: 

280 """ 

281 コーナーキック時の探索アクション。フロンティア探索ではなく、コーナーを分担して確認する。 

282 

283 フィールドに存在する(情報共有が新しい)ロボットのIDを昇順に並べ、 

284 その順番で左右のコーナーを交互に割り当てる。 

285 立ち位置の基準は担当するコーナーと同じ側のゴールエリアの角で、 

286 同じコーナーを見るロボットが複数いる場合は、IDの順番に応じた 

287 オフセット分だけそこからコーナーの反対側へずらすので、立ち位置が重ならない。 

288 """ 

289 # 立ち位置に到着したとみなす閾値。 

290 # 到着後はその場旋回し続けるので向きは判定に使わない。向きも見ると 

291 # 旋回で必ず閾値を外れ、move_to_target_global_pose の姿勢合わせと 

292 # 旋回が交互に出て振動してしまう。 

293 CORNERKICK_ARRIVE_TOLERANCE_CM = 50.0 

294 print_in_real_action("Executing cornerkick_exploration().") 

295 if not action_functions.need_flontier_exploration(s): 

296 # そもそも探索不要なら何もしない 

297 logger.action_precondition_not_satisfied( 

298 "No need to perform cornerkick exploration" 

299 ) 

300 return None 

301 

302 if s.world_states["set_play"] != SET_PLAY.CORNER_KICK: 

303 logger.action_precondition_not_satisfied( 

304 "Not in corner kick set play, cannot perform cornerkick exploration" 

305 ) 

306 raise StrategyContractError( 

307 "cornerkick_exploration called outside of corner kick set play" 

308 ) 

309 

310 watch_ids = get_cornerkick_watch_robot_ids(s) 

311 if s.robot_id not in watch_ids: 

312 # 自分がフィールドにいる扱いになっていない(想定外)ので何もしない 

313 logger.action_precondition_not_satisfied( 

314 f"robot_id={s.robot_id} is not in cornerkick watch ids {watch_ids}" 

315 ) 

316 return None 

317 

318 order = watch_ids.index(s.robot_id) 

319 # 偶数番目は左(y+)、奇数番目は右(y-)のコーナーを担当する 

320 corner_y_sign = 1.0 if order % 2 == 0 else -1.0 

321 corner_label = "LEFT" if corner_y_sign > 0 else "RIGHT" 

322 # 同じコーナー担当の中でのID順。これがコーナーからの追加オフセット段数になる 

323 offset_step = order // 2 

324 

325 target_pose = get_cornerkick_watch_pose(s, corner_y_sign, offset_step) 

326 

327 if ( 

328 calculate_distance_global(s.self_position_global, target_pose) 

329 <= CORNERKICK_ARRIVE_TOLERANCE_CM 

330 ): 

331 # 到着したら、その場をゆっくり旋回しながら首でボールを探す。 

332 # 立ち位置を離れずに視野をコーナー周辺全体へ広げるため。 

333 pub.set_neck_search_ball_mode() 

334 pub.send_walk_command( 

335 WalkCommand(0.0, 0.0, theta_velocity=_cornerkick_search_theta_velocity(s)) 

336 ) 

337 logger.action_executed( 

338 details=( 

339 f"Watching {corner_label} corner " 

340 f"(offset_step={offset_step}, turning in place to search)" 

341 ) 

342 ) 

343 return s 

344 

345 cmd = action_functions.move_to_target_global_pose(s, pub, target_pose) 

346 if cmd is not None: 

347 pub.send_walk_command(cmd) 

348 logger.action_executed( 

349 details=( 

350 f"Watching {corner_label} corner " 

351 f"(order={order}/{len(watch_ids)}, offset_step={offset_step}, " 

352 f"target=({target_pose.x:.0f}, {target_pose.y:.0f}))" 

353 ) 

354 ) 

355 return s
