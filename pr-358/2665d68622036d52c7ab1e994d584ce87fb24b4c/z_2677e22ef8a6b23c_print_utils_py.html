42def _resolve_strategy_log_dir() -> str: 

43 """ 

44 ログ出力先ディレクトリを決定する。 

45 STRATEGY_LOG_DIRで明示指定された場合はそれを最優先する(実機で集約先を変えたい場合用)。 

46 指定がなければROS 2のログディレクトリ(ROS_LOG_DIR > ROS_HOME/log > ~/.ros/log)の配下に置く。 

47 launch側のログと同じ場所にまとまり、かつsudoなしで書けるパスになるのが狙い。 

48 """ 

49 override = os.environ.get("STRATEGY_LOG_DIR") 

50 if override: 

51 return override 

52 ros_log_dir = os.environ.get("ROS_LOG_DIR") 

53 if not ros_log_dir: 

54 ros_home = os.environ.get("ROS_HOME") or os.path.expanduser("~/.ros") 

55 ros_log_dir = os.path.join(ros_home, "log") 

56 return os.path.join(ros_log_dir, "rfc_tsudanuma")

59STRATEGY_LOG_DIR = _resolve_strategy_log_dir() 

60STRATEGY_LOG_PATH = os.path.join(STRATEGY_LOG_DIR, "strategy.log") 

61# 書き込みごとのwrite(2)を避けるためのバッファサイズ。これが埋まった時だけフラッシュされる。 

62STRATEGY_LOG_BUFFER_SIZE = 1 << 16

65def _open_strategy_log_file() -> IO[bytes] | None: 

66 """ 

67 ストラテジログファイルをモジュール読み込み時に一度だけ開く。 

68 エンコード処理を挟まないバイナリの追記モードで、大きめのバッファを持たせて開く。 

69 開けなかった場合はNoneを返し、ログ出力は無効化される(実行そのものは止めない)。 

70 """ 

71 try: 

72 os.makedirs(STRATEGY_LOG_DIR, exist_ok=True) 

73 return open(STRATEGY_LOG_PATH, "ab", buffering=STRATEGY_LOG_BUFFER_SIZE) 

74 except OSError as error: 

75 sys.stderr.write( 

76 f"{STRATEGY_PREFIX} \033[31m[ERROR]\033[0m " 

77 f"failed to open {STRATEGY_LOG_PATH}: {error}\n" 

78 ) 

79 return None

85def _close_strategy_log_file() -> None: 

86 """ 

87 プロセス終了時にバッファをフラッシュしてファイルを閉じる。 

88 """ 

89 global _strategy_log_file 

90 log_file = _strategy_log_file 

91 _strategy_log_file = None 

92 if log_file is not None: 

93 try: 

94 log_file.close() 

95 except OSError: 

96 pass

99def _handle_termination_signal(signal_number: int, frame: FrameType | None) -> None: 

100 """ 

101 SIGTERMで殺された場合はatexitが走らないため、ここでフラッシュしてから 

102 デフォルト動作(プロセス終了)に委ねる。 

103 """ 

104 _close_strategy_log_file() 

105 signal.signal(signal_number, signal.SIG_DFL) 

106 os.kill(os.getpid(), signal_number) 

107 

108 

109if _strategy_log_file is not None: 

110 atexit.register(_close_strategy_log_file) 

111 try: 

112 # 既にアプリ側でハンドラが設定されている場合は上書きしない。 

113 if signal.getsignal(signal.SIGTERM) is signal.SIG_DFL: 

114 signal.signal(signal.SIGTERM, _handle_termination_signal) 

115 except ValueError: 

116 # メインスレッド以外からのimportではシグナルハンドラを設定できない。 

117 pass

120def print_to_strategy_log_file(message: str) -> None: 

121 """ 

122 Print the message to the strategy log file. 

123 STRATEGY_LOG_PATH(既定は~/.ros/log/rfc_tsudanuma/strategy.log)に出力する。 

124 行末に[strategy_log: 時刻]のサフィックスを付ける。 

125 ファイルオープンは遅いので、この関数では行わない。ファイルは既に開いている前提とする。 

126 バッファが埋まった時にだけ書き出され、明示的なフラッシュは行わない。 

127 """ 

128 log_file = _strategy_log_file 

129 if log_file is None: 

130 return 

131 timestamp = datetime.now().strftime("%H:%M:%S.%f")[:-3] 

132 log_file.write(f"{message} [strategy_log: {timestamp}]\n".encode())

135def caller_function_name(default: str = "unknown") -> str: 

136 """この関数を呼んだ関数の、さらに1つ外側の関数名を返す。 

137 

138 つまり `print_in_real_action()` から呼べば、`print_in_real_action()` を 

139 呼んだアクション関数の名前が返る。 

140 

141 `inspect.currentframe()` は Python スタックフレームを持たない処理系では 

142 None を返しうる (CPython では常に非 None)。取得できない場合は `default`。 

143 """ 

144 frame = inspect.currentframe() 

145 try: 

146 caller = frame.f_back if frame is not None else None 

147 target = caller.f_back if caller is not None else None 

148 return target.f_code.co_name if target is not None else default 

149 finally: 

150 # フレームの循環参照を残さないよう明示的に破棄する 

151 del frame
